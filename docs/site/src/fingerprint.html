1# بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
2"""Decide what a router's answer means - without guessing.
3 
4The old version had two ways to judge a reply:
5 
6    1. byte-compare after masking  -> failed as soon as the page carried a
7       session token / echoed card / timestamp the mask did not know about
8    2. "if the length differs from the failure page by more than 40 bytes,
9       call it a DIFFERENT page and stop as a suspected match"
10 
11Rule 2 is why wrong cards were reported as hits (see kirapass_hits.txt in the
12history: eight wrong cards, all "DIFFERENT page len=8575"), and why the tool
13could not recognise a *real* success that redirected to the internet.
14 
15This module does it the way a network engineer would:
16 
17    * send two deliberately-wrong probes and learn WHICH parts of the page
18      change on their own (tokens, mac echo, timestamps ...) -> mask them
19    * a reply identical to that masked rejection page = REJECTED
20    * a reply is only ACCEPTED on positive evidence (redirect out of the
21      portal, learned success words, or a real internet check)
22    * anything else = UNKNOWN, reported honestly instead of being called a hit
23"""  # نهاية النص متعدد الأسطر

34TOKEN_RE = re.compile(r"[A-Za-z0-9_@.\-]+|[^\sA-Za-z0-9_]")  # إسناد نتيجة استدعاء re.compile (معامل واحد) إلى TOKEN_RE
35SCRIPT_RE = re.compile(r"<(script|style)\b.*?</\1>", re.S | re.I)  # إسناد نتيجة استدعاء re.compile (2 معاملات) إلى SCRIPT_RE
36TAG_RE = re.compile(r"<[^>]+>")  # إسناد نتيجة استدعاء re.compile (معامل واحد) إلى TAG_RE
37WS_RE = re.compile(r"[ \t\r\n]+")  # إسناد نتيجة استدعاء re.compile (معامل واحد) إلى WS_RE
38 
39UUID_RE = re.compile(r"\b[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-"  # إسناد نتيجة استدعاء re.compile (معامل واحد) إلى UUID_RE
40                     r"[0-9a-fA-F]{4}-[0-9a-fA-F]{12}\b")  # تكملة السطر السابق داخل القوس
41LONGHEX_RE = re.compile(r"\b[0-9a-fA-F]{12,}\b")  # إسناد نتيجة استدعاء re.compile (معامل واحد) إلى LONGHEX_RE
42LONGNUM_RE = re.compile(r"\b\d{4,}\b")  # إسناد نتيجة استدعاء re.compile (معامل واحد) إلى LONGNUM_RE
43MAC_RE = re.compile(r"\b(?:[0-9a-fA-F]{2}[:-]){5}[0-9a-fA-F]{2}\b")  # إسناد نتيجة استدعاء re.compile (معامل واحد) إلى MAC_RE
44# رموز تبدو جلسة/nonce: 6 محارف فأكثر من hex/alnum تخلط حروفاً وأرقاماً
45# («faded» أو «0201242548» لا تُخفى، و«5f30507b» تُخفى).
46TOKENLIKE_RE = re.compile(r"[0-9A-Za-z]{6,}")  # إسناد نتيجة استدعاء re.compile (معامل واحد) إلى TOKENLIKE_RE
47HEXLIKE_RE = re.compile(r"^[0-9a-fA-F]+$")  # إسناد نتيجة استدعاء re.compile (معامل واحد) إلى HEXLIKE_RE
48IP_RE = re.compile(r"\b\d{1,3}(?:\.\d{1,3}){3}\b")  # إسناد نتيجة استدعاء re.compile (معامل واحد) إلى IP_RE

54def visible_text(body: str) -> str:  # تعريف الدالة visible_text(body) ترجع str
55    """What the guest actually sees: no <script>/<style>, no tags."""  # نص توثيقي (docstring) يشرح ما يليه
56    if not body:  # شرط معكوس: ليس body
57        return ""  # إرجاع ''
58    text = SCRIPT_RE.sub(" ", body)  # إسناد نتيجة استدعاء SCRIPT_RE.sub (2 معاملات) إلى text
59    text = TAG_RE.sub(" ", text)  # إسناد نتيجة استدعاء TAG_RE.sub (2 معاملات) إلى text
60    text = html.unescape(text)  # إسناد نتيجة استدعاء html.unescape (معامل واحد) إلى text
61    return WS_RE.sub(" ", text).strip()  # إرجاع WS_RE.sub(' ', text).strip()

68def _nonce_like(tok: str) -> bool:  # تعريف الدالة _nonce_like(tok) ترجع bool
69    if len(tok) < 6:  # شرط: len(tok) أصغر من 6
70        return False  # إرجاع False
71    has_digit = any(c.isdigit() for c in tok)  # إسناد نتيجة استدعاء any (معامل واحد) إلى has_digit
72    has_alpha = any(c.isalpha() for c in tok)  # إسناد نتيجة استدعاء any (معامل واحد) إلى has_alpha
73    return has_digit and has_alpha  # إرجاع has_digit و has_alpha

76# يستبدل القيم المتغيّرة والأنماط المتعلَّمة بـ# حتى تصير المقارنة عادلة.
77def mask_text(text: str, literals=(), generic: bool = True, patterns=()) -> str:  # تعريف الدالة mask_text(text, literals, generic, patterns) ترجع str
78    """Replace values that legitimately change between two identical requests."""  # نص توثيقي (docstring) يشرح ما يليه
79    out = text or ""  # دمج منطقي (أو) وإسناده إلى out
80    for lit in literals:  # دورة على literals باسم lit
81        if lit and len(str(lit)) >= 3:  # شرط مركّب (و)
82            out = out.replace(str(lit), "#")  # إسناد نتيجة استدعاء out.replace (2 معاملات) إلى out
83    for pattern in patterns:  # دورة على patterns باسم pattern
84        try:  # بدايةtry محمية (يليها except/finally)
85            out = re.sub(pattern, "#", out)  # إسناد نتيجة استدعاء re.sub (3 معاملات) إلى out
86        except re.error:  # تكملة السطر السابق داخل القوس
87            continue  # الانتقال إلى الدورة التالية
88    if generic:  # شرط: generic
89        out = TOKENLIKE_RE.sub(lambda m: "#" if _nonce_like(m.group(0))  # إسناد نتيجة استدعاء TOKENLIKE_RE.sub (2 معاملات) إلى out
90                               else m.group(0), out)  # تكملة السطر السابق داخل القوس
91        out = UUID_RE.sub("#", out)  # إسناد نتيجة استدعاء UUID_RE.sub (2 معاملات) إلى out
92        out = LONGHEX_RE.sub("#", out)  # إسناد نتيجة استدعاء LONGHEX_RE.sub (2 معاملات) إلى out
93        out = MAC_RE.sub("#", out)  # إسناد نتيجة استدعاء MAC_RE.sub (2 معاملات) إلى out
94        out = IP_RE.sub("#", out)  # إسناد نتيجة استدعاء IP_RE.sub (2 معاملات) إلى out
95        out = LONGNUM_RE.sub("#", out)  # إسناد نتيجة استدعاء LONGNUM_RE.sub (2 معاملات) إلى out
96    return out  # إرجاع out

99# «شكل» الصفحة: تسلسل الرموز بعد تحويل كل رقم إلى # — للمقارنة التقريبية.
100def struct_tokens(text: str, literals=(), generic: bool = True, patterns=()) -> list:  # تعريف الدالة struct_tokens(text, literals, generic, patterns) ترجع list
101    """A shape-only view of the page, used when exact compare is impossible."""  # نص توثيقي (docstring) يشرح ما يليه
102    masked = mask_text(text, literals, generic, patterns)  # إسناد نتيجة استدعاء mask_text (4 معاملات) إلى masked
103    out = []  # إسناد قائمة إلى out
104    for tok in tokens(masked):  # دورة على tokens(masked) باسم tok
105        if tok.isdigit():  # شرط: نتيجة tok.isdigit()
106            out.append("#")  # استدعاء out.append (معامل واحد)
107        else:  # مفتاح else في القاموس
108            out.append(tok.lower())  # استدعاء out.append (معامل واحد)
109    return out  # إرجاع out

118def diff_words(page: str, reference: str, limit: int = 12) -> dict:  # تعريف الدالة diff_words(page, reference, limit) ترجع dict
119    """Words that appear in `page` but not in `reference`, and vice-versa."""  # نص توثيقي (docstring) يشرح ما يليه
120    def words(t):  # تعريف الدالة words(t)
121        return set(re.findall(r"[A-Za-z\u0600-\u06FF]{3,}", visible_text(t).lower()))  # إرجاع set(re.findall('[A-Za-z\\u0600-\\u06FF]{3,}', visible_text(t).lower()))
122 
123    pw, rw = words(page), words(reference)  # إسناد مجموعة إلى مجموعة
124    return {  # إرجاع قاموس
125        "new_words": sorted(pw - rw, key=len, reverse=True)[:limit],  # مفتاح new_words في القاموس
126        "missing_words": sorted(rw - pw, key=len, reverse=True)[:limit],  # مفتاح missing_words في القاموس
127    }  # إغلاق القوس المفتوح في السطر السابق

166def differing_tokens(a: str, b: str, limit: int = 40) -> list:  # تعريف الدالة differing_tokens(a, b, limit) ترجع list
167    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
168    """Raw list of the values that change between two same-shape replies.
169 
170    Reported to the user ("this page changes by itself in N places"); the
171    masking engine below is what actually neutralises them.
172    """  # نهاية النص متعدد الأسطر
173    if not a or not b:  # شرط مركّب (أو)
174        return []  # إرجاع قائمة
175    sm = difflib.SequenceMatcher(None, a, b, autojunk=False)  # إسناد نتيجة استدعاء difflib.SequenceMatcher (3 معاملات، autojunk=…) إلى sm
176    out, seen = [], set()  # إسناد مجموعة إلى مجموعة
177    for tag, i1, i2, j1, j2 in sm.get_opcodes():  # دورة على sm.get_opcodes() باسم مجموعة
178        if tag == "equal":  # شرط: tag يساوي 'equal'
179            continue  # الانتقال إلى الدورة التالية
180        for piece in (a[i1:i2], b[j1:j2]):  # دورة على مجموعة باسم piece
181            for tok in tokens(piece):  # دورة على tokens(piece) باسم tok
182                if len(tok) >= 3 and tok not in seen:  # شرط مركّب (و)
183                    seen.add(tok)  # استدعاء seen.add (معامل واحد)
184                    out.append(tok)  # استدعاء out.append (معامل واحد)
185                    if len(out) >= limit:  # شرط: len(out) أكبر أو يساوي limit
186                        return out  # إرجاع out
187    return out  # إرجاع out

190# يحوّل القيم المتغيّرة إلى أنماط عامة بدل حفظ كل قيمة على حدة.
191def learn_patterns(samples, max_patterns: int = 6) -> list:  # تعريف الدالة learn_patterns(samples, max_patterns) ترجع list
192    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
193    """Turn "this span changes" into "a token like this changes".
194 
195    A session token of 8 hex characters is different on every reply, so
196    remembering the value is useless.  Seeing that the value LOOKED like
197    `[0-9a-fA-F]{8}` is what lets us ignore it forever.
198    """  # نهاية النص متعدد الأسطر
199    samples = [x for x in samples if x]  # بناء اشتقاق قائمة وإسناده إلى samples
200    if len(samples) < 2:  # شرط: len(samples) أصغر من 2
201        return []  # إرجاع قائمة
202    out, seen = [], set()  # إسناد مجموعة إلى مجموعة
203    base = samples[0]  # إسناد samples[0] إلى base
204    for other in samples[1:]:  # دورة على samples[] باسم other
205        sm = difflib.SequenceMatcher(None, base, other, autojunk=False)  # إسناد نتيجة استدعاء difflib.SequenceMatcher (3 معاملات، autojunk=…) إلى sm
206        for tag, i1, i2, j1, j2 in sm.get_opcodes():  # دورة على sm.get_opcodes() باسم مجموعة
207            if tag == "equal":  # شرط: tag يساوي 'equal'
208                continue  # الانتقال إلى الدورة التالية
209            for piece in (base[i1:i2], other[j1:j2]):  # دورة على مجموعة باسم piece
210                for tok in tokens(piece):  # دورة على tokens(piece) باسم tok
211                    if not _nonce_like(tok) or len(tok) > 64:  # شرط مركّب (أو)
212                        continue  # الانتقال إلى الدورة التالية
213                    if HEXLIKE_RE.match(tok):  # شرط: نتيجة HEXLIKE_RE.match(tok)
214                        pattern = r"\b[0-9a-fA-F]{%d}\b" % len(tok)  # حساب باقي القسمة بين '\\b[0-9a-fA-F]{%d}\\b' وlen(tok) وإسناده إلى pattern
215                    else:  # مفتاح else في القاموس
216                        pattern = r"\b[0-9A-Za-z]{%d}\b" % len(tok)  # حساب باقي القسمة بين '\\b[0-9A-Za-z]{%d}\\b' وlen(tok) وإسناده إلى pattern
217                    if pattern not in seen:  # شرط: pattern ليس ضمن seen
218                        seen.add(pattern)  # استدعاء seen.add (معامل واحد)
219                        out.append(pattern)  # استدعاء out.append (معامل واحد)
220                        if len(out) >= max_patterns:  # شرط: len(out) أكبر أو يساوي max_patterns
221                            return out  # إرجاع out
222    return out  # إرجاع out

225# يكتشف القيم التي تتغير وحدها بين ردّين متطابقين (رمز جلسة، وقت، صدى البطاقة).
226def learn_dynamic(samples, rounds: int = 4) -> set:  # تعريف الدالة learn_dynamic(samples, rounds) ترجع set
227    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
228    """Return the literal values that differ between two same-shape replies.
229 
230    Two wrong cards sent in a row normally produce the same page, except for
231    the session token, the echoed card, timestamps... Those are exactly the
232    values we must ignore, and we learn them instead of listing them by hand.
233    """  # نهاية النص متعدد الأسطر
234    literals = set()  # إسناد نتيجة استدعاء set إلى literals
235    samples = [s for s in samples if s]  # بناء اشتقاق قائمة وإسناده إلى samples
236    if len(samples) < 2:  # شرط: len(samples) أصغر من 2
237        return literals  # إرجاع literals
238 
239    for _ in range(rounds):  # دورة على range(rounds) باسم _
240        masked = [mask_text(s, literals) for s in samples]  # بناء اشتقاق قائمة وإسناده إلى masked
241        if all(m == masked[0] for m in masked):  # شرط: نتيجة all(مولّد)
242            break  # قطع الحلقة فوراً
243        base = masked[0]  # إسناد masked[0] إلى base
244        gained = False  # إسناد القيمة الثابتة gained
245        for other in samples[1:]:  # دورة على samples[] باسم other
246            sm = difflib.SequenceMatcher(None, base, other, autojunk=False)  # إسناد نتيجة استدعاء difflib.SequenceMatcher (3 معاملات، autojunk=…) إلى sm
247            for tag, i1, i2, j1, j2 in sm.get_opcodes():  # دورة على sm.get_opcodes() باسم مجموعة
248                if tag == "equal":  # شرط: tag يساوي 'equal'
249                    continue  # الانتقال إلى الدورة التالية
250                for piece in (base[i1:i2], other[j1:j2]):  # دورة على مجموعة باسم piece
251                    for tok in tokens(piece):  # دورة على tokens(piece) باسم tok
252                        # شرط مركّب (أو)
253                        if len(tok) >= 3 and not tok.isdigit() or \
254                                (tok.isdigit() and len(tok) >= 5):  # تكملة تعريف متعدد الأسطر
255                            if tok not in literals:  # شرط: tok ليس ضمن literals
256                                literals.add(tok)  # استدعاء literals.add (معامل واحد)
257                                gained = True  # إسناد القيمة الثابتة gained
258        if not gained:  # شرط معكوس: ليس gained
259            break  # قطع الحلقة فوراً
260    return literals  # إرجاع literals

266# حكم واحد: code + reason + confidence + data + evidence.
267# is_hit: أي كود يبدأ بـACCEPTED. is_final: مؤكد أو ثقة ≥0.85.
268class Verdict:  # تعريف الصنف Verdict
269    """One answer about one reply."""  # نص توثيقي (docstring) يشرح ما يليه
270 
271    __slots__ = ("code", "confidence", "reason", "data", "evidence")  # إسناد مجموعة إلى __slots__
272 
273    def __init__(self, code, reason, confidence=0.0, data=None, evidence=None):  # تعريف الدالة __init__(self, code, reason, confidence, data, evidence)
274        self.code = code  # إسناد code إلى self.code
275        self.reason = reason  # إسناد reason إلى self.reason
276        self.confidence = confidence  # إسناد confidence إلى self.confidence
277        self.data = data or {}  # دمج منطقي (أو) وإسناده إلى self.data
278        self.evidence = evidence or {}  # دمج منطقي (أو) وإسناده إلى self.evidence
279 
280    @property  # مُزخرف (decorator) بـproperty
281    def is_hit(self) -> bool:  # تعريف الدالة is_hit(self) ترجع bool

317    @classmethod  # مُزخرف (decorator) بـclassmethod

321        """`reject_replies` = list of Replies to wrong-but-well-formed cards."""  # نص توثيقي (docstring) يشرح ما يليه

326        bodies = [r.text for r in replies]  # بناء اشتقاق قائمة وإسناده إلى bodies
327        if login_reply is not None:  # شرط: login_reply ليس نفسه None

330            fp.note = "no rejection baseline"  # إسناد القيمة الثابتة fp.note
331            return fp  # إرجاع fp
332 
333        fp.samples = len(bodies)  # إسناد نتيجة استدعاء len (معامل واحد) إلى fp.samples

335        fp.reject_len = len(bodies[0])  # إسناد نتيجة استدعاء len (معامل واحد) إلى fp.reject_len
336        fp.reject_text = bodies[0][:40000]  # إسناد bodies[0][] إلى fp.reject_text
337        # قابل للاستخدام فقط عندما كانت كل عيّنة تحويلاً إلى نفس المكان: إن

340        if replies and all(r.is_redirect() for r in replies):  # شرط مركّب (و)
341            locs = {_loc_key(r.location) for r in replies}  # بناء اشتقاق مجموعة وإسناده إلى locs

354            fp.note = "exact comparison ready"  # إسناد القيمة الثابتة fp.note
355        else:  # مفتاح else في القاموس

357                                             patterns=fp.patterns)  # المعامل المسمّى patterns
358            fp.note = "page changes on its own - comparing by shape"  # إسناد القيمة الثابتة fp.note
359        return fp  # إرجاع fp

361    # -- المقارنة -------------------------------------------------------
362    # المقارنة بثلاث درجات: exact (SHA بعد الإخفاء) ← shape ← similar.
363    # الردّان الفارغان لا يعنيان تطابقاً: يُقارن مفتاح التحويل ثم الحالة.
364    def same_as_reject(self, resp, extra_literals=()) -> tuple:  # تعريف الدالة same_as_reject(self, resp, extra_literals) ترجع tuple
365        # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
366        """-> (is_rejected, how, similarity)
367 
368        `extra_literals` are the values we just sent (card, password, dst):
369        a portal that echoes them back must not look like a different page.
370        """  # نهاية النص متعدد الأسطر
371        body = resp.text or ""  # دمج منطقي (أو) وإسناده إلى body
372        literals = tuple(self.literals) + tuple(extra_literals or ())  # حساب جمع بين tuple(self.literals) وtuple(extra_literals أو مجموعة) وإسناده إلى literals
373        if not body and not self.reject_text:  # شرط مركّب (و)
374            # كلا الردّين صفحتان فارغتان — وهذا وحده لا يعني شيئاً. الراوتر
375            # الذي يحوّل عند بطاقة خاطئة يعيدنا إلى صفحة الدخول؛
376            # والبطاقة التي تعمل تذهب إلى مكان آخر. تجاهل ذلك كان
377            # يخفي كل إصابة حقيقية خلف «مثل صفحة الرفض».
378            if self.reject_location_key and resp.is_redirect():  # شرط مركّب (و)
379                same = _loc_key(resp.location) == self.reject_location_key  # مقارنة (يساوي) وإسناد النتيجة المنطقية إلى same
380                return same, "redirect", 1.0 if same else 0.0  # إرجاع مجموعة
381            if self.reject_status and resp.status != self.reject_status:  # شرط مركّب (و)
382                return False, "empty_status_differs", 0.0  # إرجاع مجموعة
383            return True, "empty", 1.0  # إرجاع مجموعة
384 
385        masked = mask_text(body, literals, patterns=self.patterns)  # إسناد نتيجة استدعاء mask_text (2 معاملات، patterns=…) إلى masked
386        if self.exact:  # شرط: self.exact
387            if sha(masked) == self.reject_masked_sha:  # شرط: sha(masked) يساوي self.reject_masked_sha
388                return True, "exact", 1.0  # إرجاع مجموعة
389        if self.reject_tokens:  # شرط: self.reject_tokens
390            sim = ratio(struct_tokens(body, literals, patterns=self.patterns),  # إسناد نتيجة استدعاء ratio (2 معاملات) إلى sim
391                        self.reject_tokens)  # تكملة السطر السابق داخل القوس
392            if sim >= 0.995:  # شرط: sim أكبر أو يساوي 0.995
393                return True, "shape", sim  # إرجاع مجموعة
394            return False, "shape", sim  # إرجاع مجموعة
395 
396        # لا خط أساس دقيق: نرجع إلى الحالة + تشابه الشكل
397        sim = ratio(struct_tokens(body, literals, patterns=self.patterns),  # إسناد نتيجة استدعاء ratio (2 معاملات) إلى sim
398                    struct_tokens(self.reject_text, literals,  # عنصر في القائمة/المعاملات (يتبعه المزيد)
399                                  patterns=self.patterns))  # المعامل المسمّى patterns
400        if resp.status == self.reject_status and sim >= 0.97:  # شرط مركّب (و)
401            return True, "similar", sim  # إرجاع مجموعة
402        return False, "similar", sim  # إرجاع مجموعة

431    # -- مساعدات ---------------------------------------------------------
432    @staticmethod  # مُزخرف (decorator) بـstaticmethod

440    @staticmethod  # مُزخرف (decorator) بـstaticmethod

445    # -- القرار ----------------------------------------------------
446    # قلب الأداة: ردّ واحد ⇒ حكم واحد.
447    # الترتيب هو القرار: (1) الحجب/الكابتشا (2) مطابقة صفحة الرفض
448    # (3) الأدلة الإيجابية: تحويل خارج البوابة، كلمات نجاح، رابط نجاح
449    # (4) UNKNOWN لكل ما لا يُثبت. لا يُسمّى شيء نجاحاً بلا دليل.
450    def classify(self, resp, submitted=None) -> Verdict:  # تعريف الدالة classify(self, resp, submitted) ترجع Verdict
451        body = resp.text or ""  # دمج منطقي (أو) وإسناده إلى body
452        text = visible_text(body)  # إسناد نتيجة استدعاء visible_text (معامل واحد) إلى text
453        low = text.lower()  # إسناد نتيجة استدعاء text.lower إلى low
454        raw_low = body.lower()  # إسناد نتيجة استدعاء body.lower إلى raw_low
455        loc = resp.location  # إسناد resp.location إلى loc
456        sub = [str(s) for s in (submitted or []) if s]  # بناء اشتقاق قائمة وإسناده إلى sub
457 
458        # --- الراوتر يحجبنا (يُفحص أولاً: إن كان خط الأساس قد
459        # تُعلّم ونحن محجوبون أصلاً، فقد تبدو صفحة الحجب
460        # رفضاً عادياً) ----------------------------------
461        # دمج منطقي (أو) وإسناده إلى ban
462        ban = self._has_phrase(raw_low, config.BAN_WORDS) or \
463            self._has_phrase(low, config.BAN_WORDS)  # تكملة السطر السابق داخل القوس
464        if ban or resp.status in (403, 429, 503):  # شرط مركّب (أو)
465            if resp.status == 429 or (ban and ("rate limit" in ban  # شرط مركّب (أو)
466                                               or "slow down" in ban)):  # تكملة تعريف متعدد الأسطر
467                return Verdict("RATE_LIMITED", "ban_page" if ban else "http_429",  # إرجاع Verdict('RATE_LIMITED', 'ban_page' إن ban وإلا 'http_429', 0.9, data=قاموس, evidence=resp.as_dict())
468                               0.9, data={"word": ban, "status": resp.status},  # عنصر في القائمة/المعاملات (يتبعه المزيد)
469                               evidence=resp.as_dict())  # المعامل المسمّى evidence
470            if resp.status == 503 and not ban:  # شرط مركّب (و)
471                # «service unavailable» راوتر/RADIUS يغرق —
472                # وإخبار المستخدم أنه محجوب سيرسله ليعيد تشغيل
473                # الراوتر بلا سبب. تُعامل كتقييد طلبات: أبطئ،
474                # وتوقّف إن استمر.
475                return Verdict("RATE_LIMITED", "http_503", 0.9,  # إرجاع Verdict('RATE_LIMITED', 'http_503', 0.9, data=قاموس, evidence=resp.as_dict())
476                               data={"status": resp.status},  # المعامل المسمّى data
477                               evidence=resp.as_dict())  # المعامل المسمّى evidence
478            return Verdict("BANNED", "ban_page" if ban else f"http_{resp.status}",  # إرجاع Verdict('BANNED', 'ban_page' إن ban وإلا نص منسّق (f-string), 0.9, data=قاموس, evidence=resp.as_dict())
479                           0.9, data={"word": ban, "status": resp.status},  # عنصر في القائمة/المعاملات (يتبعه المزيد)
480                           evidence=resp.as_dict())  # المعامل المسمّى evidence
481 
482        rejected, how, sim = self.fp.same_as_reject(resp, extra_literals=sub)  # إسناد نتيجة استدعاء self.fp.same_as_reject (معامل واحد، extra_literals=…) إلى مجموعة
483        if rejected and how != "redirect" and resp.is_redirect():  # شرط مركّب (و)
484            # لا يمكن أن يكون الردّ «مثل صفحة الرفض» ويخرج بالمتصفح
485            # من البوابة في الوقت نفسه: الراوتر يردّ بشيء آخر
486            # على هذه البطاقة. تسمية ذلك رفضاً عادياً هي كيف تضيع
487            # بطاقة عاملة، لذلك تذهب إلى المراجعة
488            # — مع السبب والوجهة مكتوبين.
489            host = _host(loc)  # إسناد نتيجة استدعاء _host (معامل واحد) إلى host
490            if host and host != self.portal_host:  # شرط مركّب (و)
491                return Verdict("UNKNOWN",  # إرجاع Verdict('UNKNOWN', 'redirect_out_of_portal_but_page_matches', 0.0, data=قاموس, evidence=resp.as_dict())
492                               "redirect_out_of_portal_but_page_matches",  # عنصر في القائمة/المعاملات (يتبعه المزيد)
493                               0.0,  # عنصر في القائمة/المعاملات (يتبعه المزيد)
494                               data={"location": loc[:200], "how": how,  # المعامل المسمّى data
495                                     "similarity": round(sim, 3)},  # مفتاح similarity في القاموس
496                               evidence=resp.as_dict())  # المعامل المسمّى evidence
497        if rejected:  # شرط: rejected
498            seen_ok = [w for w in self.success_words if w in low or w in raw_low]  # بناء اشتقاق قائمة وإسناده إلى seen_ok
499            if seen_ok and how != "exact":  # شرط مركّب (و)
500                # الصفحة تطابق شكل الرفض، لكن كلمات تعلّمناها
501                # من بطاقة عاملة موجودة ⇒ دَع إنساناً ينظر إليها
502                return Verdict("UNKNOWN", "looks_rejected_but_success_words_found",  # إرجاع Verdict('UNKNOWN', 'looks_rejected_but_success_words_found', 0.0, data=قاموس, evidence=resp.as_dict())
503                               0.0, data={"words": seen_ok[:6]},  # عنصر في القائمة/المعاملات (يتبعه المزيد)
504                               evidence=resp.as_dict())  # المعامل المسمّى evidence
505            return Verdict("REJECTED", f"same_as_rejection_page_{how}", 1.0,  # إرجاع Verdict('REJECTED', نص منسّق (f-string), 1.0, data=قاموس, evidence=resp.as_dict())
506                           data={"similarity": round(sim, 3)},  # المعامل المسمّى data
507                           evidence=resp.as_dict())  # المعامل المسمّى evidence
508 
509        # --- أُضيف كابتشا / تحد ---------------------------
510        if any(w in raw_low for w in ("captcha", "g-recaptcha", "hcaptcha")):  # شرط: نتيجة any(مولّد)
511            return Verdict("CHALLENGE", "captcha_present", 0.6,  # إرجاع Verdict('CHALLENGE', 'captcha_present', 0.6, evidence=resp.as_dict())
512                           evidence=resp.as_dict())  # المعامل المسمّى evidence
513 
514        # --- دليل إيجابي: البوابة أخرجتنا ---------------------
515        if resp.is_redirect():  # شرط: نتيجة resp.is_redirect()
516            host = _host(loc)  # إسناد نتيجة استدعاء _host (معامل واحد) إلى host
517            if host and host != self.portal_host:  # شرط مركّب (و)
518                marker = self._has_any(loc.lower(), self.exit_markers)  # إسناد نتيجة استدعاء self._has_any (2 معاملات) إلى marker
519                if not marker and self._has_any(loc.lower(),  # شرط مركّب (و)
520                                                config.PORTAL_URL_WORDS):  # تكملة تعريف متعدد الأسطر
521                    # صفحة أخرى من نفس البوابة — ليس خروجاً، وليس
522                    # شيئاً يجوز أن نسميه بطاقة عاملة أيضاً
523                    return Verdict("UNKNOWN", "redirect_to_another_portal_page",  # إرجاع Verdict('UNKNOWN', 'redirect_to_another_portal_page', 0.0, data=قاموس, evidence=resp.as_dict())
524                                   0.0, data={"location": loc[:200]},  # عنصر في القائمة/المعاملات (يتبعه المزيد)
525                                   evidence=resp.as_dict())  # المعامل المسمّى evidence
526                confident = 0.95 if marker else 0.85  # إسناد 0.95 إن marker وإلا 0.85 إلى confident
527                return Verdict("ACCEPTED", "redirect_out_of_portal", confident,  # إرجاع Verdict('ACCEPTED', 'redirect_out_of_portal', confident, data=قاموس, evidence=resp.as_dict())
528                               data={"location": loc[:200]},  # المعامل المسمّى data
529                               evidence=resp.as_dict())  # المعامل المسمّى evidence
530            # الراوتر يرسلنا إلى مكان غير الذي يرسل إليه بطاقة
531            # خاطئة — هذا دليل حقيقي، حتى لو كانت الوجهة صفحة
532            # حالته هو (لا يكفي وحده أبداً لإيقاف تشغيل)
533            if self.fp.reject_location_key and resp.is_redirect():  # شرط مركّب (و)
534                if _loc_key(loc) != self.fp.reject_location_key:  # شرط: _loc_key(loc) لا يساوي self.fp.reject_location_key
535                    return Verdict("ACCEPTED", "redirect_differs_from_rejection",  # إرجاع Verdict('ACCEPTED', 'redirect_differs_from_rejection', 0.7, data=قاموس, evidence=resp.as_dict())
536                                   0.7, data={"location": loc[:200]},  # عنصر في القائمة/المعاملات (يتبعه المزيد)
537                                   evidence=resp.as_dict())  # المعامل المسمّى evidence
538            if self.success_url_contains and self.success_url_contains in loc.lower():  # شرط مركّب (و)
539                return Verdict("ACCEPTED", "success_url_contains", 0.9,  # إرجاع Verdict('ACCEPTED', 'success_url_contains', 0.9, data=قاموس, evidence=resp.as_dict())
540                               data={"location": loc[:200]},  # المعامل المسمّى data
541                               evidence=resp.as_dict())  # المعامل المسمّى evidence
542 
543        # --- دليل إيجابي: كلمات نجاح متعلَّمة ---------------------
544        seen = [w for w in self.success_words if w in low or w in raw_low]  # بناء اشتقاق قائمة وإسناده إلى seen
545        if len(seen) >= 1 and not self._has_phrase(low, self.reject_phrases):  # شرط مركّب (و)
546            return Verdict("ACCEPTED", "learned_success_words",  # إرجاع Verdict('ACCEPTED', 'learned_success_words', min(0.9, جمع), data=قاموس, evidence=resp.as_dict())
547                           min(0.9, 0.6 + 0.1 * len(seen)),  # عنصر في القائمة/المعاملات (يتبعه المزيد)
548                           data={"words": seen[:6]}, evidence=resp.as_dict())  # المعامل المسمّى data
549 
550        if self.success_url_contains and self.success_url_contains in resp.url.lower():  # شرط مركّب (و)
551            return Verdict("ACCEPTED", "success_url_contains", 0.9,  # إرجاع Verdict('ACCEPTED', 'success_url_contains', 0.9, data=قاموس, evidence=resp.as_dict())
552                           data={"url": resp.url[:200]}, evidence=resp.as_dict())  # المعامل المسمّى data
553 
554        # --- كلمات إيجابية خاصة بصفحة *جيدة* -----------
555        accept_hit = self._has_phrase(low, self.accept_phrases)  # إسناد نتيجة استدعاء self._has_phrase (2 معاملات) إلى accept_hit
556        reject_hit = self._has_phrase(low, self.reject_phrases)  # إسناد نتيجة استدعاء self._has_phrase (2 معاملات) إلى reject_hit
557        if accept_hit and not reject_hit:  # شرط مركّب (و)
558            return Verdict("ACCEPTED", "welcome_words", 0.7,  # إرجاع Verdict('ACCEPTED', 'welcome_words', 0.7, data=قاموس, evidence=resp.as_dict())
559                           data={"word": accept_hit}, evidence=resp.as_dict())  # المعامل المسمّى data
560 
561        # --- ليست صفحة الرفض، وليست نجاحاً مثبتاً --------------
562        diff = diff_words(body, self.fp.reject_text)  # إسناد نتيجة استدعاء diff_words (2 معاملات) إلى diff
563        if reject_hit:  # شرط: reject_hit
564            # صفحة رفض متغيّرة لا نستطيع مقارنتها بدقة
565            return Verdict("REJECTED", "rejection_wording", 0.75,  # إرجاع Verdict('REJECTED', 'rejection_wording', 0.75, data=قاموس, evidence=resp.as_dict())
566                           data={"word": reject_hit}, evidence=resp.as_dict())  # المعامل المسمّى data
567 
568        return Verdict("UNKNOWN", "reply_differs_not_proven", 0.0,  # إرجاع Verdict('UNKNOWN', 'reply_differs_not_proven', 0.0, data=قاموس, evidence=resp.as_dict())
569                       data={"diff": diff, "similarity": round(sim, 3),  # المعامل المسمّى data
570                             "status": resp.status, "length": resp.length},  # مفتاح status في القاموس
571                       evidence=resp.as_dict())  # المعامل المسمّى evidence

574def _loc_key(url: str) -> tuple:  # تعريف الدالة _loc_key(url) ترجع tuple
575    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
576    """The stable part of a redirect target: scheme + host + path.
577 
578    A router that answers "wrong card" with a redirect usually appends a
579    changing token or the guest's own dst to the query string - comparing the
580    whole URL would call every reply different.
581    """  # نهاية النص متعدد الأسطر
582    from urllib.parse import urlsplit  # استيراد urlsplit من الوحدة urllib.parse
583    try:  # بدايةtry محمية (يليها except/finally)
584        parts = urlsplit(url or "")  # إسناد نتيجة استدعاء urlsplit (معامل واحد) إلى parts
585    except Exception:  # تكملة السطر السابق داخل القوس
586        return ()  # إرجاع مجموعة
587    if not parts.netloc and not parts.path:  # شرط مركّب (و)
588        return ()  # إرجاع مجموعة
589    return (parts.scheme, parts.netloc.lower(), parts.path or "/")  # إرجاع مجموعة

592def _host(url: str) -> str:  # تعريف الدالة _host(url) ترجع str
593    from urllib.parse import urlsplit  # استيراد urlsplit من الوحدة urllib.parse
594    try:  # بدايةtry محمية (يليها except/finally)
595        return (urlsplit(url).hostname or "").lower()  # إرجاع urlsplit(url).hostname أو ''.lower()
596    except Exception:  # تكملة السطر السابق داخل القوس
597        return ""  # إرجاع ''
