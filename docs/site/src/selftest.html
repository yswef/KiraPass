69def run_engine(profile: dict, attempts: int, threads: int = 3,  # تعريف الدالة run_engine(profile, attempts, threads, timeout, checks, reset, **kw)
70               timeout: float = 90.0, checks=None, reset: bool = True, **kw):  # مفتاح timeout في القاموس
71    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
72    """Fresh, throw-away run: no resume state, nothing saved on disk.
73 
74    `reset=False` keeps `space_pos`/`walk_*` - that is how the "continue where
75    you stopped" path is tested (the UI sends them back with the profile).
76    """  # نهاية النص متعدد الأسطر
77    _RUN_ID[0] += 1  # تحديث _RUN_ID[0] بعملية جمع
78    profile = dict(profile)  # إسناد نتيجة استدعاء dict (معامل واحد) إلى profile
79    profile.update({"name": f"selftest-{_RUN_ID[0]}"})  # استدعاء profile.update (معامل واحد)
80    if reset:  # شرط: reset
81        profile.update({"space_pos": 0, "walk_a": 0, "walk_b": 0})  # استدعاء profile.update (معامل واحد)
82    eng = engine.Engine(store.Store(), persist=False, checks=checks)  # إسناد نتيجة استدعاء engine.Engine (معامل واحد، persist=…، checks=…) إلى eng
83    res = eng.start(profile, attempts=attempts, threads=threads, **kw)  # إسناد نتيجة استدعاء eng.start (معامل واحد، attempts=…، threads=…، **=…) إلى res
84    if not res.get("ok"):  # شرط معكوس: ليس res.get('ok')
85        return {"state": "rejected", "error": res.get("error"),  # إرجاع مجموعة
86                "problems": res.get("problems", [])}, eng  # مفتاح problems في القاموس
87    deadline = time.time() + timeout  # حساب جمع بين time.time() وtimeout وإسناده إلى deadline
88    while eng.state in ("calibrating", "running", "stopping"):  # حلقة ما دام مقارنة
89        if time.time() > deadline:  # شرط: time.time() أكبر من deadline
90            eng.stop("timeout")  # استدعاء eng.stop (معامل واحد)
91            break  # قطع الحلقة فوراً
92        time.sleep(0.05)  # استدعاء time.sleep (معامل واحد)
93    if eng.thread is not None:  # شرط: eng.thread ليس نفسه None
94        eng.thread.join(timeout=5)  # استدعاء eng.thread.join (timeout=…)
95    return eng.status(), eng  # إرجاع مجموعة

112def t_calibration():  # تعريف الدالة t_calibration()
113    """The rejection baseline must be learned even with a changing token."""  # نص توثيقي (docstring) يشرح ما يليه
114    with MockPortal(valid_cards={"0299"}, dynamic=True,  # سياق مُدار: MockPortal(valid_cards=مجموعة فريدة, dynamic=True, pass_mode='empty') باسم portal
115                    pass_mode="empty") as portal:  # المعامل المسمّى pass_mode
116        info = scan(portal.url)  # إسناد نتيجة استدعاء scan (معامل واحد) إلى info

122        detail = (f"baseline {'exact' if fp and fp.exact else 'shaped'}, "  # بناء نص منسّق وإسناده إلى detail
123                  f"dynamic tokens={len(fp.literals) if fp else 0}, "  # تكملة السطر السابق داخل القوس
124                  f"status={fp.reject_status if fp else '-'}, "  # تكملة السطر السابق داخل القوس
125                  f"bytes={fp.reject_len if fp else '-'}")  # تكملة السطر السابق داخل القوس
