75def free_port(host: str, wanted: int, tries: int = 25) -> int:  # تعريف الدالة free_port(host, wanted, tries) ترجع int
76    for candidate in range(wanted, wanted + tries):  # دورة على range(wanted, جمع) باسم candidate
77        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:  # سياق مُدار: socket.socket(socket.AF_INET, socket.SOCK_STREAM) باسم s
78            s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)  # استدعاء s.setsockopt (3 معاملات)
79            try:  # بدايةtry محمية (يليها except/finally)
80                s.bind((host, candidate))  # استدعاء s.bind (معامل واحد)
81                return candidate  # إرجاع candidate
82            except OSError:  # تكملة السطر السابق داخل القوس
83                continue  # الانتقال إلى الدورة التالية
84    return wanted  # إرجاع wanted

171    shown = {"url": "", "lan": ""}  # إسناد قاموس إلى shown

173    def on_ready(real_port, real_token):  # تعريف الدالة on_ready(real_port, real_token)
174        base_local = f"http://127.0.0.1:{real_port}/"  # بناء نص منسّق وإسناده إلى base_local
