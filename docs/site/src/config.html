79def _env_float(name: str, default: float) -> float:  # تعريف الدالة _env_float(name, default) ترجع float
80    raw = os.environ.get(name, "")  # إسناد نتيجة استدعاء os.environ.get (2 معاملات) إلى raw
81    if not raw:  # شرط معكوس: ليس raw
82        return default  # إرجاع default
83    try:  # بدايةtry محمية (يليها except/finally)
84        value = float(raw)  # إسناد نتيجة استدعاء float (معامل واحد) إلى value
85    except ValueError:  # تكملة السطر السابق داخل القوس
86        return default  # إرجاع default
87    return max(0.5, min(value, 120.0))  # إرجاع max(0.5, min(value, 120.0))

115# بوابة الأسر تردّ على هذه بتحويل إلى صفحة الدخول.
116# وعندما تعمل البطاقة نحصل على الرد المتوقع بدلاً من ذلك.
117#
118# بعض الشبكات تحجب مواقع الفحص الثلاثة عمداً. عندها ضع
119# روابط فحصك في متغير بيئة، والمداخل مفصولة بفواصل:
120#
121# KIRAPASS_INTERNET_CHECKS="http://example.com/generate_204|204"
122# KIRAPASS_INTERNET_CHECKS="http://a.test/ok|200|OK,http://b.test/204|204"
123#
124# الصيغة:  الرابط|الحالة المتوقعة|النص المتوقع (اختياري)
125def _parse_internet_checks(raw: str) -> tuple:  # تعريف الدالة _parse_internet_checks(raw) ترجع tuple
126    out = []  # إسناد قائمة إلى out
127    for item in (raw or "").split(","):  # دورة على raw أو ''.split(',') باسم item
128        item = item.strip()  # إسناد نتيجة استدعاء item.strip إلى item
129        if not item:  # شرط معكوس: ليس item
130            continue  # الانتقال إلى الدورة التالية
131        bits = item.split("|")  # إسناد نتيجة استدعاء item.split (معامل واحد) إلى bits
132        url = bits[0].strip()  # إسناد نتيجة استدعاء bits[0].strip إلى url
133        if not url:  # شرط معكوس: ليس url
134            continue  # الانتقال إلى الدورة التالية
135        try:  # بدايةtry محمية (يليها except/finally)
136            status = int(bits[1]) if len(bits) > 1 and bits[1].strip() else 204  # إسناد int(bits[1]) إن مقارنة و bits[1].strip() وإلا 204 إلى status
137        except ValueError:  # تكملة السطر السابق داخل القوس
138            status = 204  # إسناد القيمة الثابتة status
139        text = bits[2].strip() if len(bits) > 2 and bits[2].strip() else None  # إسناد bits[2].strip() إن مقارنة و bits[2].strip() وإلا None إلى text
140        out.append((url, status, text, "custom"))  # استدعاء out.append (معامل واحد)
141    return tuple(out)  # إرجاع tuple(out)
