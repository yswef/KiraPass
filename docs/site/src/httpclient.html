2"""A small HTTP client built only on the python standard library.
3 
4Why not `requests`?  Because the old version died with
5`ModuleNotFoundError: No module named requests` on any machine where the
6package was not installed (phones, fresh Windows, locked lab PCs).  This
7client needs nothing but python itself, and it gives exact control over the
8three things this tool lives on:
9 
10* keep-alive connection reuse  -> real speed on a router
11* connect timeout vs read timeout -> we can tell "unreachable" from "busy"
12* reading the real reply body   -> we need the bytes, not a convenience layer
13"""  # نهاية النص متعدد الأسطر

60    def decode(self, limit: int = 0) -> str:  # تعريف الدالة decode(self, limit) ترجع str
61        raw = self._body  # إسناد self._body إلى raw
62        if limit:  # شرط: limit
63            raw = raw[:limit]  # إسناد raw[] إلى raw
64        charset = "utf-8"  # إسناد القيمة الثابتة charset
65        ctype = self.header("content-type")  # إسناد نتيجة استدعاء self.header (معامل واحد) إلى ctype
66        if "charset=" in ctype:  # شرط: 'charset=' ضمن ctype
67            charset = ctype.split("charset=", 1)[1].split(";")[0].strip() or "utf-8"  # دمج منطقي (أو) وإسناده إلى charset
68        try:  # بدايةtry محمية (يليها except/finally)
69            return raw.decode(charset, "replace")  # إرجاع raw.decode(charset, 'replace')
70        except LookupError:  # تكملة السطر السابق داخل القوس
71            return raw.decode("utf-8", "replace")  # إرجاع raw.decode('utf-8', 'replace')

97class CookieStore:  # تعريف الصنف CookieStore
98    """Minimal cookie jar (name/domain/path), enough for captive portals."""  # نص توثيقي (docstring) يشرح ما يليه
99 
100    def __init__(self):  # تعريف الدالة __init__(self)
101        self._jar = {}  # إسناد قاموس إلى self._jar

103    # يحفظ Set-Cookie مع النطاق والمسار وsecure حتى لا يُرسل كوكي في غير مكانه.
104    def update(self, url: str, set_cookie_headers) -> None:  # تعريف الدالة update(self, url, set_cookie_headers) ترجع None
105        host = urllib.parse.urlsplit(url).hostname or ""  # دمج منطقي (أو) وإسناده إلى host
106        for raw in set_cookie_headers:  # دورة على set_cookie_headers باسم raw
107            try:  # بدايةtry محمية (يليها except/finally)
108                parsed = http.cookies.SimpleCookie()  # إسناد نتيجة استدعاء http.cookies.SimpleCookie إلى parsed
109                parsed.load(raw)  # استدعاء parsed.load (معامل واحد)
110            except Exception:  # تكملة السطر السابق داخل القوس
111                continue  # الانتقال إلى الدورة التالية
112            for name, morsel in parsed.items():  # دورة على parsed.items() باسم مجموعة
113                domain = (morsel["domain"] or host).lstrip(".").lower()  # إسناد نتيجة استدعاء morsel['domain'] أو host.lstrip('.').lower إلى domain
114                path = morsel["path"] or "/"  # دمج منطقي (أو) وإسناده إلى path
115                self._jar[name] = {  # إسناد قاموس إلى self._jar[name]
116                    "value": morsel.value,  # مفتاح value في القاموس
117                    "domain": domain,  # مفتاح domain في القاموس
118                    "path": path,  # مفتاح path في القاموس
119                    "secure": bool(morsel["secure"]),  # مفتاح secure في القاموس
120                }  # إغلاق القوس المفتوح في السطر السابق
121 
122    def header(self, url: str) -> str:  # تعريف الدالة header(self, url) ترجع str
123        parts = urllib.parse.urlsplit(url)  # إسناد نتيجة استدعاء urllib.parse.urlsplit (معامل واحد) إلى parts
124        host = (parts.hostname or "").lower()  # إسناد نتيجة استدعاء parts.hostname أو ''.lower إلى host
125        path = parts.path or "/"  # دمج منطقي (أو) وإسناده إلى path
126        out = []  # إسناد قائمة إلى out
127        for name, c in self._jar.items():  # دورة على self._jar.items() باسم مجموعة
128            dom = c["domain"]  # إسناد c['domain'] إلى dom
129            if dom and not (host == dom or host.endswith("." + dom)):  # شرط مركّب (و)
130                continue  # الانتقال إلى الدورة التالية
131            if not path.startswith(c["path"]):  # شرط معكوس: ليس path.startswith(c['path'])
132                continue  # الانتقال إلى الدورة التالية
133            if c["secure"] and parts.scheme != "https":  # شرط مركّب (و)
134                continue  # الانتقال إلى الدورة التالية
135            out.append(f"{name}={c['value']}")  # استدعاء out.append (معامل واحد)
136        return "; ".join(out)  # إرجاع '; '.join(out)
137 
138    def clear(self) -> None:  # تعريف الدالة clear(self) ترجع None
139        self._jar.clear()  # استدعاء self._jar.clear

153    def __init__(self, headers=None, connect_timeout=config.CONNECT_TIMEOUT,  # تعريف الدالة __init__(self, headers, connect_timeout, read_timeout, verify_tls, allow_redirects)

177    def _ssl_context(self):  # تعريف الدالة _ssl_context(self)
178        if self._ssl_ctx is None:  # شرط: self._ssl_ctx هو نفسه None
179            if self.verify_tls:  # شرط: self.verify_tls
180                self._ssl_ctx = ssl.create_default_context()  # إسناد نتيجة استدعاء ssl.create_default_context إلى self._ssl_ctx

185    def _connection(self, scheme, host, port):  # تعريف الدالة _connection(self, scheme, host, port)
186        key = (scheme, host, port)  # إسناد مجموعة إلى key
187        if self._conn is not None and self._key == key:  # شرط مركّب (و)
188            return self._conn  # إرجاع self._conn
189        self.close()  # استدعاء self.close
190        if scheme == "https":  # شرط: scheme يساوي 'https'
191            conn = http.client.HTTPSConnection(  # إسناد نتيجة استدعاء http.client.HTTPSConnection (2 معاملات، timeout=…، context=…) إلى conn
192                host, port, timeout=self.connect_timeout,  # عنصر في القائمة/المعاملات (يتبعه المزيد)
193                context=self._ssl_context())  # المعامل المسمّى context
194        else:  # مفتاح else في القاموس
195            conn = http.client.HTTPConnection(  # إسناد نتيجة استدعاء http.client.HTTPConnection (2 معاملات، timeout=…) إلى conn
196                host, port, timeout=self.connect_timeout)  # تكملة السطر السابق داخل القوس
197        self._conn, self._key = conn, key  # إسناد مجموعة إلى مجموعة
198        return conn  # إرجاع conn

200    def close(self):  # تعريف الدالة close(self)
201        if self._conn is not None:  # شرط: self._conn ليس نفسه None
202            try:  # بدايةtry محمية (يليها except/finally)
203                self._conn.close()  # استدعاء self._conn.close
204            except Exception:  # تكملة السطر السابق داخل القوس
205                pass  # سطر فارغ منطقياً (pass) — مطلوب صياغياً
206        self._conn = None  # إسناد القيمة الثابتة self._conn
207        self._key = None  # إسناد القيمة الثابتة self._key

210    # الواجهة العامة: إعادة محاولة صامتة واحدة على stale، ثم تصنيف الخطأ.
211    def request(self, method, url, params=None, data=None, headers=None,  # تعريف الدالة request(self, method, url, params, data, headers, allow_redirects, timeout) ترجع Response
212                allow_redirects=None, timeout=None) -> Response:  # المعامل المسمّى allow_redirects
213        method = method.upper()  # إسناد نتيجة استدعاء method.upper إلى method
214        if params:  # شرط: params
215            sep = "&" if urllib.parse.urlsplit(url).query else "?"  # إسناد '&' إن urllib.parse.urlsplit(url).query وإلا '?' إلى sep
216            url = url + sep + urllib.parse.urlencode(params, doseq=True)  # حساب جمع بين جمع وurllib.parse.urlencode(params, doseq=True) وإسناده إلى url
217        hdr_ct = ""  # إسناد القيمة الثابتة hdr_ct
218        if headers:  # شرط: headers
219            for key, value in headers.items():  # دورة على headers.items() باسم مجموعة
220                if str(key).lower() == "content-type":  # شرط: str(key).lower() يساوي 'content-type'
221                    hdr_ct = value or ""  # دمج منطقي (أو) وإسناده إلى hdr_ct
222                    break  # قطع الحلقة فوراً
223        if isinstance(data, dict):  # شرط: نتيجة isinstance(data, dict)
224            if "json" in hdr_ct.lower():  # شرط: 'json' ضمن hdr_ct.lower()
225                body = json.dumps(data).encode("utf-8")  # إسناد نتيجة استدعاء json.dumps(data).encode (معامل واحد) إلى body
226                post_headers = {"Content-Type": hdr_ct or "application/json"}  # إسناد قاموس إلى post_headers
227            else:  # مفتاح else في القاموس
228                body = urllib.parse.urlencode(data, doseq=True).encode()  # إسناد نتيجة استدعاء urllib.parse.urlencode(data, doseq=True).encode إلى body
229                post_headers = {"Content-Type": "application/x-www-form-urlencoded"}  # إسناد قاموس إلى post_headers
230        elif isinstance(data, str):  # شرط: نتيجة isinstance(data, str)
231            body = data.encode()  # إسناد نتيجة استدعاء data.encode إلى body
232            post_headers = {"Content-Type": hdr_ct or "application/x-www-form-urlencoded"}  # إسناد قاموس إلى post_headers
233        elif isinstance(data, bytes):  # شرط: نتيجة isinstance(data, bytes)
234            body = data  # إسناد data إلى body
235            post_headers = {"Content-Type": "application/octet-stream"}  # إسناد قاموس إلى post_headers
236        else:  # مفتاح else في القاموس
237            body = None  # إسناد القيمة الثابتة body
238            post_headers = {}  # إسناد قاموس إلى post_headers
239 
240        follow = self.allow_redirects if allow_redirects is None else allow_redirects  # إسناد self.allow_redirects إن مقارنة وإلا allow_redirects إلى follow
241        history = []  # إسناد قائمة إلى history
242        started = time.time()  # إسناد نتيجة استدعاء time.time إلى started
243 
244        for _ in range(MAX_REDIRECTS + 1):  # دورة على range(جمع) باسم _
245            resp = self._send_once(method, url, body, headers, post_headers,  # إسناد نتيجة استدعاء self._send_once (6 معاملات) إلى resp
246                                   timeout)  # تكملة السطر السابق داخل القوس
247            resp.history = list(history)  # إسناد نتيجة استدعاء list (معامل واحد) إلى resp.history
248            if not (follow and resp.is_redirect()):  # شرط معكوس: ليس follow و resp.is_redirect()
249                resp.elapsed_ms = (time.time() - started) * 1000  # حساب ضرب بين طرح و1000 وإسناده إلى resp.elapsed_ms
250                return resp  # إرجاع resp
251            history.append(url)  # استدعاء history.append (معامل واحد)
252            target = urllib.parse.urljoin(url, resp.location)  # إسناد نتيجة استدعاء urllib.parse.urljoin (2 معاملات) إلى target
253            if resp.status == 303 or (resp.status in (301, 302) and method == "POST"):  # شرط مركّب (أو)
254                method, body, post_headers = "GET", None, {}  # إسناد مجموعة إلى مجموعة
255            url = target  # إسناد target إلى url
256        raise NetError("too_many_redirects", f"redirect loop at {url}", url)  # رفع NetError('too_many_redirects', نص منسّق (f-string), url)
257

264    # -- رحلة الذهاب والإياب الخام الواحدة -------------------------------
265    # إرسال طلب واحد على اتصال موجود أو جديد.
266    # فصل مهلة الاتصال عن مهلة القراءة مقصود: بدونه يدمج بايثون المرحلتين
267    # في TimeoutError واحد فيبدو الراوتر الميت «بطيئاً» بدل «غير قابل للوصول».
268    def _send_once(self, method, url, body, extra_headers, post_headers,  # تعريف الدالة _send_once(self, method, url, body, extra_headers, post_headers, timeout) ترجع Response
269                   timeout) -> Response:  # تكملة تعريف متعدد الأسطر
270        parts = urllib.parse.urlsplit(url)  # إسناد نتيجة استدعاء urllib.parse.urlsplit (معامل واحد) إلى parts
271        if parts.scheme not in ("http", "https"):  # شرط: parts.scheme ليس ضمن مجموعة
272            raise NetError("proto", f"unsupported scheme: {parts.scheme!r}", url)  # رفع NetError('proto', نص منسّق (f-string), url)
273        host = parts.hostname or ""  # دمج منطقي (أو) وإسناده إلى host
274        if not host:  # شرط معكوس: ليس host
275            raise NetError("proto", f"no host in url: {url}", url)  # رفع NetError('proto', نص منسّق (f-string), url)
276        scheme = parts.scheme  # إسناد parts.scheme إلى scheme
277        port = parts.port or (443 if scheme == "https" else 80)  # دمج منطقي (أو) وإسناده إلى port
278        path = urllib.parse.urlunsplit(("", "", parts.path or "/", parts.query, ""))  # إسناد نتيجة استدعاء urllib.parse.urlunsplit (معامل واحد) إلى path
279 
280        headers = dict(self.headers)  # إسناد نتيجة استدعاء dict (معامل واحد) إلى headers
281        headers.update(post_headers)  # استدعاء headers.update (معامل واحد)

291                   else (timeout or self.read_timeout))  # تكملة السطر السابق داخل القوس
292        connect_to = (timeout[0] if isinstance(timeout, (tuple, list))  # إسناد timeout[0] إن isinstance(timeout, مجموعة) وإلا self.connect_timeout إلى connect_to
293                      else self.connect_timeout)  # تكملة السطر السابق داخل القوس
