42def atomic_write(path: str, text: str) -> None:  # تعريف الدالة atomic_write(path, text) ترجع None
43    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
44    """Write a file so that a reader never sees half of it.
45 
46    Tolerates the folder being deleted at the same moment (the user pressing
47    "clear cache" while a run is saving something).
48    """  # نهاية النص متعدد الأسطر
49    for attempt in range(2):  # دورة على range(2) باسم attempt
50        try:  # بدايةtry محمية (يليها except/finally)
51            ensure_dirs()  # استدعاء ensure_dirs
52            tmp = path + ".tmp"  # حساب جمع بين path و'.tmp' وإسناده إلى tmp
53            with open(tmp, "w", encoding="utf-8") as fh:  # سياق مُدار: open(tmp, 'w', encoding='utf-8') باسم fh
54                fh.write(text)  # استدعاء fh.write (معامل واحد)
55            os.replace(tmp, path)  # استدعاء os.replace (2 معاملات)
56            return  # إنهاء الدالة بلا قيمة (ترجع None ضمناً)
57        except OSError:  # تكملة السطر السابق داخل القوس
58            if attempt:  # شرط: attempt
59                return  # إنهاء الدالة بلا قيمة (ترجع None ضمناً)

132def space_size(p: dict) -> int:  # تعريف الدالة space_size(p) ترجع int
133    vlen = variable_len(p)  # إسناد نتيجة استدعاء variable_len (معامل واحد) إلى vlen
134    chars = len(set(p.get("charset", "")))  # إسناد نتيجة استدعاء len (معامل واحد) إلى chars
135    if vlen <= 0 or chars < 2:  # شرط مركّب (أو)
136        return 0  # إرجاع 0
137    return chars ** vlen  # إرجاع أسّ

142def validate(p: dict) -> list:  # تعريف الدالة validate(p) ترجع list
143    """-> list of machine-readable problems (empty = good to go)."""  # نص توثيقي (docstring) يشرح ما يليه

165# رقم ⇒ بطاقة: تحويل بأساس عدد المحارف الفريدة المرتّبة.
166# sorted(set(charset)) مقصود: أبجدية فيها تكرار تُنتج بطاقات مكرّرة.
167def decode_card(p: dict, index: int) -> str:  # تعريف الدالة decode_card(p, index) ترجع str
168    charset = p.get("charset", "")  # إسناد نتيجة استدعاء p.get (2 معاملات) إلى charset
169    vlen = variable_len(p)  # إسناد نتيجة استدعاء variable_len (معامل واحد) إلى vlen
170    base = len(set(charset))  # إسناد نتيجة استدعاء len (معامل واحد) إلى base
171    chars = sorted(set(charset))  # إسناد نتيجة استدعاء sorted (معامل واحد) إلى chars
172    out = []  # إسناد قائمة إلى out
173    for _ in range(vlen):  # دورة على range(vlen) باسم _
174        out.append(chars[index % base])  # استدعاء out.append (معامل واحد)
175        index //= base  # تحديث index بعملية قسمة صحيحة
176    return p.get("prefix", "") + "".join(out) + p.get("suffix", "")  # إرجاع جمع

179def sample_cards(p: dict, count: int = 3) -> list:  # تعريف الدالة sample_cards(p, count) ترجع list
180    space = space_size(p)  # إسناد نتيجة استدعاء space_size (معامل واحد) إلى space
181    if space <= 0:  # شرط: space أصغر أو يساوي 0
182        return []  # إرجاع قائمة
183    picks = {0, space // 2, space - 1, space // 3}  # إسناد مجموعة فريدة إلى picks
184    return [decode_card(p, i) for i in sorted(picks)][:count]  # إرجاع اشتقاق قائمة[]

187# مشي مخلوط بلا تكرار: (b + a*pos) mod space حيث a أولي نسبياً مع space.
188def card_at_walk_pos(p: dict, pos: int) -> str:  # تعريف الدالة card_at_walk_pos(p, pos) ترجع str
189    """Shuffled but repeat-free walk over the whole space."""  # نص توثيقي (docstring) يشرح ما يليه
190    space = space_size(p)  # إسناد نتيجة استدعاء space_size (معامل واحد) إلى space
191    if space <= 1:  # شرط: space أصغر أو يساوي 1
192        return decode_card(p, 0)  # إرجاع decode_card(p, 0)
193    a = int(p.get("walk_a", 0)) % space  # حساب باقي القسمة بين int(p.get('walk_a', 0)) وspace وإسناده إلى a
194    b = int(p.get("walk_b", 0)) % space  # حساب باقي القسمة بين int(p.get('walk_b', 0)) وspace وإسناده إلى b
195    if a == 0:  # شرط: a يساوي 0
196        a = 1  # إسناد القيمة الثابتة a
197    idx = (b + a * pos) % space  # حساب باقي القسمة بين جمع وspace وإسناده إلى idx
198    return decode_card(p, idx)  # إرجاع decode_card(p, idx)

204def migrate(raw: dict) -> dict:  # تعريف الدالة migrate(raw) ترجع dict
205    """Accept a profile written by ANY older version of the tool."""  # نص توثيقي (docstring) يشرح ما يليه
206    if not isinstance(raw, dict):  # شرط معكوس: ليس isinstance(raw, dict)
207        return new_profile()  # إرجاع new_profile()
208    p = new_profile()  # إسناد نتيجة استدعاء new_profile إلى p
209    p.update({k: v for k, v in raw.items() if k in p})  # استدعاء p.update (معامل واحد)
210 
211    nt = str(raw.get("network_type", "1"))  # إسناد نتيجة استدعاء str (معامل واحد) إلى nt
212    prefix = raw.get("prefix", p["prefix"]) or ""  # دمج منطقي (أو) وإسناده إلى prefix
213    suffix = raw.get("suffix", p["suffix"]) or ""  # دمج منطقي (أو) وإسناده إلى suffix
214    charset = raw.get("charset") or p["charset"]  # دمج منطقي (أو) وإسناده إلى charset
215    vlen = raw.get("var_len")  # إسناد نتيجة استدعاء raw.get (معامل واحد) إلى vlen
216    if vlen:  # شرط: vlen
217        length = int(vlen) + len(prefix) + len(suffix)  # حساب جمع بين جمع وlen(suffix) وإسناده إلى length
218    else:  # مفتاح else في القاموس
219        length = int(raw.get("length", p["length"]) or p["length"])  # إسناد نتيجة استدعاء int (معامل واحد) إلى length
220 
221    if nt == "3":  # شرط: nt يساوي '3'
222        # بروفايلات «اسم مستخدم + كلمة مرور مختلفة» القديمة: أبقِ شكل
223        # اسم المستخدم، ويصير جانب كلمة المرور قيمة ثابتة/مشتقة.
224        prefix = raw.get("u_prefix", prefix) or ""  # دمج منطقي (أو) وإسناده إلى prefix
225        suffix = raw.get("u_suffix", suffix) or ""  # دمج منطقي (أو) وإسناده إلى suffix
226        charset = raw.get("u_charset", charset) or charset  # دمج منطقي (أو) وإسناده إلى charset
227        length = int(raw.get("u_len", 0) or 0) + len(prefix) + len(suffix) or length  # دمج منطقي (أو) وإسناده إلى length
228        p["pass_mode"] = "fixed"  # إسناد القيمة الثابتة p['pass_mode']
229    elif nt == "2" and raw.get("pass_mode") in (None, "same"):  # شرط مركّب (و)
230        p["pass_mode"] = "same"  # إسناد القيمة الثابتة p['pass_mode']
231    elif p.get("pass_mode") in (None, ""):  # شرط: p.get('pass_mode') ضمن مجموعة
232        p["pass_mode"] = "empty"  # إسناد القيمة الثابتة p['pass_mode']
233 
234    p.update({  # استدعاء p.update (معامل واحد)
235        "charset": charset,  # مفتاح charset في القاموس
236        "length": max(length, len(prefix) + len(suffix) + 1),  # مفتاح length في القاموس
237        "prefix": prefix,  # مفتاح prefix في القاموس
238        "suffix": suffix,  # مفتاح suffix في القاموس
239        "method": "post" if str(raw.get("method", "2")) in ("2", "post") else "get",  # مفتاح method في القاموس
240        "send_dst": bool(raw.get("send_dst")) if "send_dst" in raw  # مفتاح send_dst في القاموس
241                    else bool(raw.get("extras", True)),  # عنصر في القائمة/المعاملات (يتبعه المزيد)
242        "send_popup": bool(raw.get("send_popup")) if "send_popup" in raw  # مفتاح send_popup في القاموس
243                      else bool(raw.get("extras", True)),  # عنصر في القائمة/المعاملات (يتبعه المزيد)
244        "dst_value": raw.get("dst_value", ""),  # مفتاح dst_value في القاموس
245        "dst_field": raw.get("dst_field") or "dst",  # مفتاح dst_field في القاموس
246        "popup_field": raw.get("popup_field") or "popup",  # مفتاح popup_field في القاموس
247        "extra_fields": raw.get("fixed_fields") or raw.get("extra_fields") or {},  # مفتاح extra_fields في القاموس
248        "success_words": raw.get("success_signature") or raw.get("success_words") or [],  # مفتاح success_words في القاموس
249        "success_url_contains": raw.get("success_url_contains", "") or "",  # مفتاح success_url_contains في القاموس
250        "space_pos": int(raw.get("space_pos", 0) or 0),  # مفتاح space_pos في القاموس
251        "space_pass": int(raw.get("space_pass", 0) or 0),  # مفتاح space_pass في القاموس
252        "walk_a": int(raw.get("walk_a", 0) or 0),  # مفتاح walk_a في القاموس
253        "walk_b": int(raw.get("walk_b", 0) or 0),  # مفتاح walk_b في القاموس
254        "schema": SCHEMA,  # مفتاح schema في القاموس
255    })  # إغلاق القوس المفتوح في السطر السابق
256    if (raw.get("walk") or {}).get("pos"):  # شرط: نتيجة raw.get('walk') أو قاموس.get('pos')
257        p["space_pos"] = int(raw["walk"]["pos"])  # إسناد نتيجة استدعاء int (معامل واحد) إلى p['space_pos']
258    # البروفايلات القديمة قد تحتوي رابط GET ناجحاً منسوخاً بالبطاقة
259    # وتجزئة كلمة المرور في استعلامه. أبقِ شكل طلبه، لا أسرار.

380    def cleanup_old_reports(self, days: int = None) -> dict:  # تعريف الدالة cleanup_old_reports(self, days) ترجع dict
381        # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
382        """Delete run reports older than `days` (default from config).
383 
384        Optional and never touches hits/profiles. Returns how many files went.
385        """  # نهاية النص متعدد الأسطر
386        retain = config.REPORT_RETENTION_DAYS if days is None else int(days)  # إسناد config.REPORT_RETENTION_DAYS إن مقارنة وإلا int(days) إلى retain
387        if retain <= 0:  # شرط: retain أصغر أو يساوي 0
388            return {"removed": 0, "freed_bytes": 0, "days": retain}  # إرجاع قاموس
389        cutoff = time.time() - retain * 86400  # حساب طرح بين time.time() وضرب وإسناده إلى cutoff
390        removed, freed = 0, 0  # إسناد مجموعة إلى مجموعة
391        try:  # بدايةtry محمية (يليها except/finally)
392            names = os.listdir(config.RUN_DIR)  # إسناد نتيجة استدعاء os.listdir (معامل واحد) إلى names
393        except OSError:  # تكملة السطر السابق داخل القوس
394            return {"removed": 0, "freed_bytes": 0, "days": retain}  # إرجاع قاموس
395        for name in names:  # دورة على names باسم name
396            if not name.endswith(".json"):  # شرط معكوس: ليس name.endswith('.json')
397                continue  # الانتقال إلى الدورة التالية
398            path = os.path.join(config.RUN_DIR, name)  # إسناد نتيجة استدعاء os.path.join (2 معاملات) إلى path
399            try:  # بدايةtry محمية (يليها except/finally)
400                if os.path.getmtime(path) < cutoff:  # شرط: os.path.getmtime(path) أصغر من cutoff
401                    size = os.path.getsize(path)  # إسناد نتيجة استدعاء os.path.getsize (معامل واحد) إلى size
402                    os.remove(path)  # استدعاء os.remove (معامل واحد)
403                    removed += 1  # تحديث removed بعملية جمع
404                    freed += size  # تحديث freed بعملية جمع
405            except OSError:  # تكملة السطر السابق داخل القوس
406                continue  # الانتقال إلى الدورة التالية
407        return {"removed": removed, "freed_bytes": freed, "days": retain}  # إرجاع قاموس

436    def clear_cache(self, scope: str = "temp") -> dict:  # تعريف الدالة clear_cache(self, scope) ترجع dict
437        # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
438        """Delete tool data by scope.
439 
440        temp     -> saved review pages + cache folders (never the hits log)
441        results  -> run reports, logs and hits.txt
442        profiles -> profiles.json
443        all      -> everything above + legacy files from the old version
444        """  # نهاية النص متعدد الأسطر
445        removed, freed = [], 0  # إسناد مجموعة إلى مجموعة
446        targets = []  # إسناد قائمة إلى targets
447        if scope in ("temp", "all"):  # شرط: scope ضمن مجموعة
448            targets += [config.CACHE_DIR, config.REVIEW_DIR]  # تحديث targets بعملية جمع
449        if scope in ("results", "all"):  # شرط: scope ضمن مجموعة
450            # سجل الإصابات من النتائج، لا من تنظيف «المؤقت»:
451            # مسح الكاش يجب ألا يحذف أبداً بطاقات وُجدت بصمت
452            targets += [config.RUN_DIR, config.LOG_DIR, config.HITS_FILE]  # تحديث targets بعملية جمع
453        always = []  # إسناد قائمة إلى always
454        if scope == "all":  # شرط: scope يساوي 'all'
455            always += list(config.LEGACY_FILES)  # تحديث always بعملية جمع
456            targets += [config.DATA_DIR]  # تحديث targets بعملية جمع
457        for path in targets + always:  # دورة على جمع باسم path
458            if not os.path.exists(path):  # شرط معكوس: ليس os.path.exists(path)
459                continue                      # لا تسرد ما لم يكن موجوداً
460            freed += _remove_path(path)  # تحديث freed بعملية جمع
461            removed.append(os.path.basename(path))  # استدعاء removed.append (معامل واحد)
462        if scope in ("profiles", "all") and os.path.exists(config.PROFILES_FILE):  # شرط مركّب (و)
463            try:  # بدايةtry محمية (يليها except/finally)
464                os.remove(config.PROFILES_FILE)  # استدعاء os.remove (معامل واحد)
465                removed.append(os.path.basename(config.PROFILES_FILE))  # استدعاء removed.append (معامل واحد)
466                self._profiles = {}  # إسناد قاموس إلى self._profiles
467            except OSError:  # تكملة السطر السابق داخل القوس
468                pass  # سطر فارغ منطقياً (pass) — مطلوب صياغياً
469        # كاشات البايت كود للأداة نفسها (نمطا glob
470        # متداخلان — بدون هذه المجموعة كان نفس المجلد يُحسب مرتين
471        # وكان رقم «المحرَّر» كذبة)
472        seen_paths = set()  # إسناد نتيجة استدعاء set إلى seen_paths
473        for pattern in ("**/__pycache__", "__pycache__"):  # دورة على مجموعة باسم pattern
474            for path in glob.glob(os.path.join(config.BASE_DIR, pattern),  # دورة على glob.glob(os.path.join(config.BASE_DIR, pattern), recursive=True) باسم path
475                                  recursive=True):  # المعامل المسمّى recursive
476                key = os.path.normpath(path)  # إسناد نتيجة استدعاء os.path.normpath (معامل واحد) إلى key
477                if key in seen_paths:  # شرط: key ضمن seen_paths
478                    continue  # الانتقال إلى الدورة التالية
479                seen_paths.add(key)  # استدعاء seen_paths.add (معامل واحد)
480                size, _ = _dir_size(path)  # إسناد نتيجة استدعاء _dir_size (معامل واحد) إلى مجموعة
481                shutil.rmtree(path, ignore_errors=True)  # استدعاء shutil.rmtree (معامل واحد، ignore_errors=…)
482                freed += size  # تحديث freed بعملية جمع
483                removed.append(os.path.relpath(path, config.BASE_DIR))  # استدعاء removed.append (معامل واحد)
484        ensure_dirs()  # استدعاء ensure_dirs
485        return {"scope": scope, "removed": sorted(set(removed)),  # إرجاع قاموس
486                "freed_bytes": freed, "freed_human": human_size(freed)}  # مفتاح freed_bytes في القاموس

489def _dir_size(path: str) -> tuple:  # تعريف الدالة _dir_size(path) ترجع tuple
490    total, count = 0, 0  # إسناد مجموعة إلى مجموعة
491    if not os.path.isdir(path):  # شرط معكوس: ليس os.path.isdir(path)
492        return 0, 0  # إرجاع مجموعة
493    for root, _dirs, files in os.walk(path):  # دورة على os.walk(path) باسم مجموعة
494        for f in files:  # دورة على files باسم f
495            try:  # بدايةtry محمية (يليها except/finally)
496                total += os.path.getsize(os.path.join(root, f))  # تحديث total بعملية جمع
497                count += 1  # تحديث count بعملية جمع
498            except OSError:  # تكملة السطر السابق داخل القوس
499                continue  # الانتقال إلى الدورة التالية
500    return total, count  # إرجاع مجموعة

503# يميّز الملف من المجلد ويرجع البايتات المحررة فعلاً.
504# shutil.rmtree على ملف لا يفعل شيئاً بصمت — هكذا كانت hits.txt تُحسب محذوفة.
505def _remove_path(path: str) -> int:  # تعريف الدالة _remove_path(path) ترجع int
506    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
507    """Delete a file or a folder and return the bytes really freed.
508 
509    `shutil.rmtree(..., ignore_errors=True)` silently does nothing when handed
510    a *file* - which is how the old code managed to list hits.txt as "removed"
511    without removing it.  This helper never lies about what it did.
512    """  # نهاية النص متعدد الأسطر
513    if os.path.isdir(path) and not os.path.islink(path):  # شرط مركّب (و)
514        size, _ = _dir_size(path)  # إسناد نتيجة استدعاء _dir_size (معامل واحد) إلى مجموعة
515        shutil.rmtree(path, ignore_errors=True)  # استدعاء shutil.rmtree (معامل واحد، ignore_errors=…)
516        return size  # إرجاع size
517    try:  # بدايةtry محمية (يليها except/finally)
518        size = os.path.getsize(path)  # إسناد نتيجة استدعاء os.path.getsize (معامل واحد) إلى size
519        os.remove(path)  # استدعاء os.remove (معامل واحد)
520        return size  # إرجاع size
521    except OSError:  # تكملة السطر السابق داخل القوس
522        return 0  # إرجاع 0

534def redact_review_body(body: str) -> str:  # تعريف الدالة redact_review_body(body) ترجع str
535    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
536    """Strip client MAC/IP addresses from review pages before saving.
537 
538    Review HTML can echo the guest device identity; keep the page shape for
539    the operator without storing those identifiers on disk.
540    """  # نهاية النص متعدد الأسطر
541    text = body or ""  # دمج منطقي (أو) وإسناده إلى text
542    text = _REVIEW_MAC_RE.sub("[mac-redacted]", text)  # إسناد نتيجة استدعاء _REVIEW_MAC_RE.sub (2 معاملات) إلى text
543    text = _REVIEW_IP_RE.sub("[ip-redacted]", text)  # إسناد نتيجة استدعاء _REVIEW_IP_RE.sub (2 معاملات) إلى text
544    return text  # إرجاع text

547def human_size(num: int) -> str:  # تعريف الدالة human_size(num) ترجع str
548    value = float(num)  # إسناد نتيجة استدعاء float (معامل واحد) إلى value
549    for unit in ("B", "KB", "MB", "GB"):  # دورة على مجموعة باسم unit
550        if value < 1024 or unit == "GB":  # شرط مركّب (أو)
551            return f"{int(value)} B" if unit == "B" else f"{value:.1f} {unit}"  # إرجاع نص منسّق (f-string) إن مقارنة وإلا نص منسّق (f-string)
552        value /= 1024.0  # تحديث value بعملية قسمة
553    return f"{value:.1f} GB"  # إرجاع نص منسّق (f-string)

556def load_legacy_profiles() -> list:  # تعريف الدالة load_legacy_profiles() ترجع list
557    """Import profiles written by the old version, if any exist."""  # نص توثيقي (docstring) يشرح ما يليه
