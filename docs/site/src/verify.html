10"""  # نهاية النص متعدد الأسطر

19def resolve_internet_checks(custom=None) -> tuple:  # تعريف الدالة resolve_internet_checks(custom) ترجع tuple
20    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
21    """Build the check list from an optional custom URL string or env default.
22 
23    `custom` accepts the same pipe format as KIRAPASS_INTERNET_CHECKS, e.g.
24    ``http://example.com/generate_204|204`` or a bare URL (expects HTTP 204).
25    """  # نهاية النص متعدد الأسطر
26    if custom is None or custom == "":  # شرط مركّب (أو)
27        return config.INTERNET_CHECKS  # إرجاع config.INTERNET_CHECKS
28    if isinstance(custom, (list, tuple)):  # شرط: نتيجة isinstance(custom, مجموعة)
29        return tuple(custom) if custom else config.INTERNET_CHECKS  # إرجاع tuple(custom) إن custom وإلا config.INTERNET_CHECKS
30    raw = str(custom).strip()  # إسناد نتيجة استدعاء str(custom).strip إلى raw
31    if not raw:  # شرط معكوس: ليس raw
32        return config.INTERNET_CHECKS  # إرجاع config.INTERNET_CHECKS
33    parsed = config._parse_internet_checks(raw)  # إسناد نتيجة استدعاء config._parse_internet_checks (معامل واحد) إلى parsed
34    return parsed or config.INTERNET_CHECKS  # إرجاع parsed أو config.INTERNET_CHECKS

37# السؤال الوحيد المهم: هل هذا الجهاز متصل فعلاً الآن؟
38# allow_redirects=False مقصود: التحويل إلى صفحة الدخول هو دليل WALLED.
39# ترجع (state, detail) حيث state ∈ {ONLINE, WALLED, OFFLINE}.
40def probe_internet(session, checks=None, timeout=(3.0, 6.0)) -> dict:  # تعريف الدالة probe_internet(session, checks, timeout) ترجع dict
41    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
42    """-> {state, label, detail, status, location}
43 
44    state: ONLINE  -> the check URL answered exactly as expected (no portal)
45           WALLED  -> we were redirected to the portal (needs a card)
46           OFFLINE -> nothing answered at all
47    """  # نهاية النص متعدد الأسطر
48    last_error = None  # إسناد القيمة الثابتة last_error
49    for url, want_status, want_text, label in (checks or config.INTERNET_CHECKS):  # دورة على checks أو config.INTERNET_CHECKS باسم مجموعة
50        try:  # بدايةtry محمية (يليها except/finally)
51            resp = session.get(url, allow_redirects=False, timeout=timeout)  # إسناد نتيجة استدعاء session.get (معامل واحد، allow_redirects=…، timeout=…) إلى resp
52        # تكملة السطر السابق داخل القوس
53        except Exception as exc:                       # noqa: BLE001
54            last_error = exc  # إسناد exc إلى last_error
55            continue  # الانتقال إلى الدورة التالية
56        body = resp.text or ""  # دمج منطقي (أو) وإسناده إلى body
57        if resp.status == want_status and (want_text is None or want_text in body):  # شرط مركّب (و)
58            return {"state": "ONLINE", "label": label, "detail": "expected_answer",  # إرجاع قاموس
59                    "status": resp.status, "url": url, "location": ""}  # مفتاح status في القاموس
60        if resp.is_redirect():  # شرط: نتيجة resp.is_redirect()
61            return {"state": "WALLED", "label": label, "detail": "portal_redirect",  # إرجاع قاموس
62                    "status": resp.status, "url": url,  # مفتاح status في القاموس
63                    "location": resp.location[:200]}  # مفتاح location في القاموس
64        if resp.status in (200, 204) and want_text and want_text not in body:  # شرط مركّب (و)
65            return {"state": "WALLED", "label": label, "detail": "portal_page",  # إرجاع قاموس
66                    "status": resp.status, "url": url,  # مفتاح status في القاموس
67                    "location": ""}  # مفتاح location في القاموس
68        if resp.status in (403, 429, 503):  # شرط: resp.status ضمن مجموعة
69            return {"state": "BLOCKED", "label": label, "detail": f"http_{resp.status}",  # إرجاع قاموس
70                    "status": resp.status, "url": url, "location": ""}  # مفتاح status في القاموس
71    # لم يردّ شيء إطلاقاً. قل *لماذا* بنفس الكلمات التي تستخدمها
72    # بقية الأداة (dns / refused / stale / tls …) حتى تستطيع الواجهة ترجمتها.
73    kind = "no_answer"  # إسناد القيمة الثابتة kind
74    if last_error is not None:  # شرط: last_error ليس نفسه None
75        try:  # بدايةtry محمية (يليها except/finally)
76            from .errors import classify  # استيراد classify من الوحدة errors
77            kind = classify(last_error, "").kind  # إسناد classify(last_error, '').kind إلى kind
78        # تكملة السطر السابق داخل القوس
79        except Exception:                              # noqa: BLE001
80            kind = type(last_error).__name__  # إسناد type(last_error).__name__ إلى kind
81    return {"state": "OFFLINE", "label": "", "detail": kind,  # إرجاع قاموس
82            "status": 0, "url": "", "location": ""}  # مفتاح status في القاموس

85def verify_online(session, checks=None, timeout=(3.0, 6.0)) -> tuple:  # تعريف الدالة verify_online(session, checks, timeout) ترجع tuple
86    """After a card is accepted: -> (True/False, dict with the reason)."""  # نص توثيقي (docstring) يشرح ما يليه
87    info = probe_internet(session, checks=checks, timeout=timeout)  # إسناد نتيجة استدعاء probe_internet (معامل واحد، checks=…، timeout=…) إلى info
88    ok = info["state"] == "ONLINE"  # مقارنة (يساوي) وإسناد النتيجة المنطقية إلى ok
89    return ok, info  # إرجاع مجموعة

92def portal_status_page(session, portal_url: str, timeout=(4.0, 8.0)) -> dict:  # تعريف الدالة portal_status_page(session, portal_url, timeout) ترجع dict
93    """Look for the router's own status/logout page as a second opinion."""  # نص توثيقي (docstring) يشرح ما يليه
94    parts = urlsplit(portal_url)  # إسناد نتيجة استدعاء urlsplit (معامل واحد) إلى parts
95    base = f"{parts.scheme}://{parts.netloc}"  # بناء نص منسّق وإسناده إلى base
96    for path in ("/status", "/status.html", "/logout"):  # دورة على مجموعة باسم path
97        url = base + path  # حساب جمع بين base وpath وإسناده إلى url
98        try:  # بدايةtry محمية (يليها except/finally)
99            resp = session.get(url, allow_redirects=False, timeout=timeout)  # إسناد نتيجة استدعاء session.get (معامل واحد، allow_redirects=…، timeout=…) إلى resp
100        # تكملة السطر السابق داخل القوس
101        except Exception as exc:                       # noqa: BLE001
102            return {"checked": False, "reason": type(exc).__name__, "url": url}  # إرجاع قاموس
103        body = (resp.text or "").lower()  # إسناد نتيجة استدعاء resp.text أو ''.lower إلى body
104        if resp.status == 200 and not resp.is_redirect():  # شرط مركّب (و)
105            logged_in = any(w in body for w in  # إسناد نتيجة استدعاء any (معامل واحد) إلى logged_in
106                            ("logout", "log out", "remaining", "uptime",  # عنصر في القائمة/المعاملات (يتبعه المزيد)
107                             "session", "bytes", "disconnect"))  # تكملة السطر السابق داخل القوس
108            login_form = ('type="password"' in body or "type='password'" in body  # دمج منطقي (أو) وإسناده إلى login_form
109                          or "name=\"password\"" in body)  # تكملة السطر السابق داخل القوس
110            return {"checked": True, "url": url, "status": resp.status,  # إرجاع قاموس
111                    "looks_logged_in": bool(logged_in and not login_form),  # مفتاح looks_logged_in في القاموس
112                    "reason": "status_page_reachable"}  # مفتاح reason في القاموس
113        if resp.is_redirect():  # شرط: نتيجة resp.is_redirect()
114            return {"checked": True, "url": url, "status": resp.status,  # إرجاع قاموس
115                    "looks_logged_in": False, "reason": "redirected_to_login",  # مفتاح looks_logged_in في القاموس
116                    "location": resp.location[:160]}  # مفتاح location في القاموس
117    return {"checked": False, "reason": "no_status_page"}  # إرجاع قاموس

122def logout(session, portal_url: str, timeout=(4.0, 8.0)) -> dict:  # تعريف الدالة logout(session, portal_url, timeout) ترجع dict
123    """Log the test card out again so a paid card is not left consumed."""  # نص توثيقي (docstring) يشرح ما يليه
124    parts = urlsplit(portal_url)  # إسناد نتيجة استدعاء urlsplit (معامل واحد) إلى parts
125    base = f"{parts.scheme}://{parts.netloc}"  # بناء نص منسّق وإسناده إلى base
126    for path in ("/logout", "/login?erase-cookie=on"):  # دورة على مجموعة باسم path
127        url = base + path  # حساب جمع بين base وpath وإسناده إلى url
128        try:  # بدايةtry محمية (يليها except/finally)
129            resp = session.get(url, allow_redirects=False, timeout=timeout)  # إسناد نتيجة استدعاء session.get (معامل واحد، allow_redirects=…، timeout=…) إلى resp
130        # تكملة السطر السابق داخل القوس
131        except Exception as exc:                       # noqa: BLE001
132            continue  # الانتقال إلى الدورة التالية
133        if resp.status in (200, 302):  # شرط: resp.status ضمن مجموعة
134            return {"ok": True, "url": url, "status": resp.status}  # إرجاع قاموس
135    return {"ok": False, "reason": "no_logout_endpoint"}  # إرجاع قاموس
