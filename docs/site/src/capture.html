130def field_shape(value: str) -> dict:  # تعريف الدالة field_shape(value) ترجع dict
131    value = "" if value is None else str(value)  # إسناد '' إن مقارنة وإلا str(value) إلى value
132    classes = []  # إسناد قائمة إلى classes
133    if not value:  # شرط معكوس: ليس value
134        classes.append("empty")  # استدعاء classes.append (معامل واحد)
135    else:  # مفتاح else في القاموس
136        if re.fullmatch(r"[0-9]+", value):  # شرط: نتيجة re.fullmatch('[0-9]+', value)
137            classes.append("digits")  # استدعاء classes.append (معامل واحد)
138        elif HEX32_RE.match(value):  # شرط: نتيجة HEX32_RE.match(value)
139            classes.append("hex32")  # استدعاء classes.append (معامل واحد)
140        elif re.fullmatch(r"[0-9a-fA-F]+", value):  # شرط: نتيجة re.fullmatch('[0-9a-fA-F]+', value)
141            classes.append("hex")  # استدعاء classes.append (معامل واحد)
142        elif re.fullmatch(r"[A-Za-z0-9]+", value):  # شرط: نتيجة re.fullmatch('[A-Za-z0-9]+', value)
143            classes.append("alnum")  # استدعاء classes.append (معامل واحد)
144        else:  # مفتاح else في القاموس
145            classes.append("other")  # استدعاء classes.append (معامل واحد)
146    return {"length": len(value), "class": classes[0]}  # إرجاع قاموس

150def redact_fields(fields) -> list:  # تعريف الدالة redact_fields(fields) ترجع list
151    out = []  # إسناد قائمة إلى out
152    if not isinstance(fields, dict):  # شرط معكوس: ليس isinstance(fields, dict)
153        return out  # إرجاع out

155        value = "" if value is None else str(value)  # إسناد '' إن مقارنة وإلا str(value) إلى value
156        item = {"name": str(name), **field_shape(value)}  # إسناد قاموس إلى item
157        item["secret_name"] = looks_secret_name(str(name))  # إسناد نتيجة استدعاء looks_secret_name (معامل واحد) إلى item['secret_name']
158        item["token_name"] = looks_token_name(str(name))  # إسناد نتيجة استدعاء looks_token_name (معامل واحد) إلى item['token_name']
159        out.append(item)  # استدعاء out.append (معامل واحد)

177def infer_pass_mode(user_before, pass_before, user_after, pass_after,  # تعريف الدالة infer_pass_mode(user_before, pass_before, user_after, pass_after, html, sent_keys) ترجع dict

179    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
180    """Learn only the patterns KiraPass can replay without a browser.
181 
182    Anything else sets needs_browser_js and must not be claimed automatable.
183    sha1(user) and sha256(user) are learned the same way as md5user when the
184    captured password hex matches those digests — no arbitrary JS is executed.
185    """  # نهاية النص متعدد الأسطر
186    sent = {str(k) for k in (sent_keys or [])}  # بناء اشتقاق مجموعة وإسناده إلى sent
187    user_b = "" if user_before is None else str(user_before)  # إسناد '' إن مقارنة وإلا str(user_before) إلى user_b
188    user_a = "" if user_after is None else str(user_after)  # إسناد '' إن مقارنة وإلا str(user_after) إلى user_a
189    pass_b = None if pass_before is None else str(pass_before)  # إسناد None إن مقارنة وإلا str(pass_before) إلى pass_b
190    pass_a = None if pass_after is None else str(pass_after)  # إسناد None إن مقارنة وإلا str(pass_after) إلى pass_a
191    user = user_a or user_b  # دمج منطقي (أو) وإسناده إلى user
192 
193    pass_field_sent = any(looks_secret_name(k) and "user" not in k.lower()  # إسناد نتيجة استدعاء any (معامل واحد) إلى pass_field_sent
194                          and "card" not in k.lower()  # تكملة السطر السابق داخل القوس
195                          and "login" not in k.lower()  # تكملة السطر السابق داخل القوس
196                          and "account" not in k.lower()  # تكملة السطر السابق داخل القوس
197                          for k in sent)  # تكملة السطر السابق داخل القوس
198    if sent and not pass_field_sent and pass_a in (None, ""):  # شرط مركّب (و)
199        return {"pass_mode": "omit", "needs_browser_js": False,  # إرجاع قاموس
200                "reason": "omit", "reason_ar": "حقل كلمة المرور لم يُرسل."}  # مفتاح reason في القاموس
201 
202    chap = None  # إسناد القيمة الثابتة chap
203    form = portals.parse_form(html or "", "http://capture.invalid/")  # إسناد نتيجة استدعاء portals.parse_form (2 معاملات) إلى form
204    if form and form.chap:  # شرط مركّب (و)
205        chap = form.chap  # إسناد form.chap إلى chap
206 
207    if chap and pass_a and HEX32_RE.match(pass_a):  # شرط مركّب (و)
208        cid, chal = chap.get("id") or "", chap.get("challenge") or ""  # إسناد مجموعة إلى مجموعة
209        if cid or chal:  # شرط مركّب (أو)
210            raw_candidates = []  # إسناد قائمة إلى raw_candidates
211            if pass_b is not None:  # شرط: pass_b ليس نفسه None
212                raw_candidates.append(("typed", pass_b))  # استدعاء raw_candidates.append (معامل واحد)
213            raw_candidates.append(("user", user))  # استدعاء raw_candidates.append (معامل واحد)
214            raw_candidates.append(("empty", ""))  # استدعاء raw_candidates.append (معامل واحد)
215            for label, raw in raw_candidates:  # دورة على raw_candidates باسم مجموعة
216                if pass_a.lower() == _md5(f"{cid}{raw}{chal}"):  # شرط: pass_a.lower() يساوي _md5(نص منسّق (f-string))
217                    if label == "empty" or raw == "":  # شرط مركّب (أو)
218                        mode = "chap_empty"  # إسناد القيمة الثابتة mode
219                    else:  # مفتاح else في القاموس
220                        mode = "chap"  # إسناد القيمة الثابتة mode
221                    return {"pass_mode": mode, "needs_browser_js": False,  # إرجاع قاموس
222                            "reason": "mikrotik_chap",  # مفتاح reason في القاموس
223                            "reason_ar": "MikroTik CHAP (hexMD5) معروف وقابل للأتمتة."}  # مفتاح reason_ar في القاموس
224

226        return {"pass_mode": "md5user", "needs_browser_js": False,  # إرجاع قاموس

234                "reason": "sha1user",  # مفتاح reason في القاموس
