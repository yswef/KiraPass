53# خطأ شبكة واحد: kind (كود ثابت) + text (رسالة النظام) + url.
54# retryable وshort خاصيتان محسوبتان من جدول KINDS.
55class NetError(Exception):  # تعريف الصنف NetError يرث من Exception
56    """One network failure with a machine code and a human hint."""  # نص توثيقي (docstring) يشرح ما يليه
57 
58    def __init__(self, kind: str, text: str, url: str = ""):  # تعريف الدالة __init__(self, kind, text, url)
59        super().__init__(text)  # استدعاء super().__init__ (معامل واحد)
60        self.kind = kind  # إسناد kind إلى self.kind
61        self.text = text  # إسناد text إلى self.text
62        self.url = url  # إسناد url إلى self.url
63 
64    @property  # مُزخرف (decorator) بـproperty
65    def retryable(self) -> bool:  # تعريف الدالة retryable(self) ترجع bool
66        return KINDS.get(self.kind, KINDS["unknown"])[0]  # إرجاع KINDS.get(self.kind, KINDS['unknown'])[0]
67 
68    @property  # مُزخرف (decorator) بـproperty
69    def short(self) -> str:  # تعريف الدالة short(self) ترجع str
70        return KINDS.get(self.kind, KINDS["unknown"])[1]  # إرجاع KINDS.get(self.kind, KINDS['unknown'])[1]
71 
72    def as_dict(self) -> dict:  # تعريف الدالة as_dict(self) ترجع dict
73        return {"kind": self.kind, "text": self.text[:300], "url": self.url}  # إرجاع قاموس

76# يحوّل أي استثناء بايثون إلى NetError واحد له كود ونص وتلميح.
77# ترتيب الفروع مقصود: الأنواع الدقيقة أولاً، ثم مطابقة النص.
78# في بايثون ≥3.10 socket.timeout هو TimeoutError نفسه ويرتفع للمرحلتين،
79# لذلك مرحلة فتح الاتصال ملفوفة في httpclient._send_once وحدها.
80def classify(exc: BaseException, url: str = "") -> NetError:  # تعريف الدالة classify(exc, url) ترجع NetError
81    """Map a python exception to a NetError."""  # نص توثيقي (docstring) يشرح ما يليه
82    if isinstance(exc, NetError):  # شرط: نتيجة isinstance(exc, NetError)
83        return exc  # إرجاع exc
84 
85    text = f"{type(exc).__name__}: {exc}"  # بناء نص منسّق وإسناده إلى text
86    low = text.lower()  # إسناد نتيجة استدعاء text.lower إلى low
87 
88    if isinstance(exc, ssl.SSLError):  # شرط: نتيجة isinstance(exc, ssl.SSLError)
89        kind = "tls"  # إسناد القيمة الثابتة kind
90    elif isinstance(exc, socket.gaierror):  # شرط: نتيجة isinstance(exc, socket.gaierror)
91        kind = "dns"  # إسناد القيمة الثابتة kind
92    elif isinstance(exc, (http.client.RemoteDisconnected,)):  # شرط: نتيجة isinstance(exc, مجموعة)
93        # اتصال keep-alive أغلقه الراوتر بين طلبين
94        kind = "stale"  # إسناد القيمة الثابتة kind
95    elif isinstance(exc, (http.client.IncompleteRead,  # شرط: نتيجة isinstance(exc, مجموعة)
96                          http.client.LineTooLong)):  # تكملة تعريف متعدد الأسطر
97        kind = "bad_response"  # إسناد القيمة الثابتة kind
98    elif isinstance(exc, socket.timeout):  # شرط: نتيجة isinstance(exc, socket.timeout)
99        # بايثون ≥ 3.10: socket.timeout هو TimeoutError نفسه، وhttp.client
100        # يرفعه للمرحلتين. العميل يلفّ مرحلة الاتصال
101        # بنفسه (انظر httpclient.Session._send_once)، لذلك أي شيء
102        # يصل إلى هنا كان فعلاً قراءة لم تكتمل.
103        kind = "read_timeout"  # إسناد القيمة الثابتة kind
104    elif isinstance(exc, ConnectionRefusedError):  # شرط: نتيجة isinstance(exc, ConnectionRefusedError)
105        kind = "refused"  # إسناد القيمة الثابتة kind
106    elif isinstance(exc, (ConnectionResetError, BrokenPipeError)):  # شرط: نتيجة isinstance(exc, مجموعة)
107        kind = "reset"  # إسناد القيمة الثابتة kind
108    elif isinstance(exc, ConnectionAbortedError):  # شرط: نتيجة isinstance(exc, ConnectionAbortedError)
109        kind = "reset"  # إسناد القيمة الثابتة kind
110    elif isinstance(exc, OSError) and getattr(exc, "errno", None) == errno.ENETUNREACH:  # شرط مركّب (و)
111        kind = "unreachable"  # إسناد القيمة الثابتة kind
112    elif isinstance(exc, http.client.HTTPException):  # شرط: نتيجة isinstance(exc, http.client.HTTPException)
113        kind = "bad_response"  # إسناد القيمة الثابتة kind
114    # شرط مركّب (أو)
115    elif "remotedisconnected" in low or "remote end closed" in low \
116            or "badstatusline" in low or "cannot read from timed out" in low:  # تكملة تعريف متعدد الأسطر
117        kind = "stale"  # إسناد القيمة الثابتة kind
118    elif "reset by peer" in low or "connection aborted" in low:  # شرط مركّب (أو)
119        kind = "reset"  # إسناد القيمة الثابتة kind
120    elif "refused" in low:  # شرط: 'refused' ضمن low
121        kind = "refused"  # إسناد القيمة الثابتة kind
122    elif "timed out" in low or "timeout" in low:  # شرط مركّب (أو)
123        kind = "read_timeout"  # إسناد القيمة الثابتة kind
124    # شرط مركّب (أو)
125    elif "name or service not known" in low or "nodename nor servname" in low \
126            or "getaddrinfo" in low or "name resolution" in low:  # تكملة تعريف متعدد الأسطر
127        kind = "dns"  # إسناد القيمة الثابتة kind
128    elif "no route to host" in low or "network is unreachable" in low:  # شرط مركّب (أو)
129        kind = "unreachable"  # إسناد القيمة الثابتة kind
130    elif "incompleteread" in low or "chunked" in low or "invalid http" in low:  # شرط مركّب (أو)
131        kind = "bad_response"  # إسناد القيمة الثابتة kind
132    elif isinstance(exc, ValueError):  # شرط: نتيجة isinstance(exc, ValueError)
133        kind = "proto"  # إسناد القيمة الثابتة kind
134    else:  # مفتاح else في القاموس
135        kind = "unknown"  # إسناد القيمة الثابتة kind
136 
137    return NetError(kind, text, url)  # إرجاع NetError(kind, text, url)
