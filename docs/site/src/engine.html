38def password_value(p: dict, card: str):  # تعريف الدالة password_value(p, card)
39    """Return the password value, or None when the field must be omitted."""  # نص توثيقي (docstring) يشرح ما يليه
40    mode = p.get("pass_mode", "empty")  # إسناد نتيجة استدعاء p.get (2 معاملات) إلى mode
41    if mode == "same":  # شرط: mode يساوي 'same'
42        return card  # إرجاع card
43    if mode == "empty":  # شرط: mode يساوي 'empty'
44        return ""  # إرجاع ''
45    if mode == "omit":  # شرط: mode يساوي 'omit'
46        return None  # إرجاع None
47    if mode == "fixed":  # شرط: mode يساوي 'fixed'
48        return p.get("pass_fixed", "")  # إرجاع p.get('pass_fixed', '')
49    if mode == "md5user":  # شرط: mode يساوي 'md5user'
50        return hashlib.md5(card.encode()).hexdigest()  # إرجاع hashlib.md5(card.encode()).hexdigest()
51    if mode == "sha1user":  # شرط: mode يساوي 'sha1user'
52        return hashlib.sha1(card.encode()).hexdigest()  # إرجاع hashlib.sha1(card.encode()).hexdigest()
53    if mode == "sha256user":  # شرط: mode يساوي 'sha256user'
54        return hashlib.sha256(card.encode()).hexdigest()  # إرجاع hashlib.sha256(card.encode()).hexdigest()
55    if mode in ("chap", "chap_empty"):  # شرط: mode ضمن مجموعة
56        raw = card if mode == "chap" else ""  # إسناد card إن مقارنة وإلا '' إلى raw
57        chap = p.get("chap") or {}  # دمج منطقي (أو) وإسناده إلى chap
58        blob = f"{chap.get('id', '')}{raw}{chap.get('challenge', '')}"  # بناء نص منسّق وإسناده إلى blob
59        return hashlib.md5(blob.encode()).hexdigest()  # إرجاع hashlib.md5(blob.encode()).hexdigest()
60    return card  # إرجاع card

63def build_fields(p: dict, card: str) -> dict:  # تعريف الدالة build_fields(p, card) ترجع dict
64    fields = {}  # إسناد قاموس إلى fields
65    pw = password_value(p, card)  # إسناد نتيجة استدعاء password_value (2 معاملات) إلى pw
66    if pw is not None:  # شرط: pw ليس نفسه None
67        fields[p.get("pass_field") or "password"] = pw  # إسناد pw إلى fields[p.get('pass_field') أو ']
68    fields[p.get("user_field") or "username"] = card  # إسناد card إلى fields[p.get('user_field') أو ']
69    if p.get("send_dst") and p.get("dst_field"):  # شرط مركّب (و)
70        fields[p["dst_field"]] = p.get("dst_value", "")  # إسناد نتيجة استدعاء p.get (2 معاملات) إلى fields[p['dst_field']]
71    if p.get("send_popup") and p.get("popup_field"):  # شرط مركّب (و)
72        fields[p["popup_field"]] = "true"  # إسناد القيمة الثابتة fields[p['popup_field']]
73    special_fields = {p.get("user_field") or "username",  # إسناد مجموعة فريدة إلى special_fields
74                      p.get("pass_field") or "password",  # عنصر في القائمة/المعاملات (يتبعه المزيد)
75                      p.get("dst_field") or "dst",  # عنصر في القائمة/المعاملات (يتبعه المزيد)
76                      p.get("popup_field") or "popup"}  # تكملة السطر السابق داخل القوس
77    for key, value in (p.get("extra_fields") or {}).items():  # دورة على p.get('extra_fields') أو قاموس.items() باسم مجموعة
78        if key in fields or key in special_fields:  # شرط مركّب (أو)
79            continue  # الانتقال إلى الدورة التالية
80        fields[key] = value  # إسناد value إلى fields[key]
81    return fields  # إرجاع fields

84def _without_query_fields(url: str, names) -> str:  # تعريف الدالة _without_query_fields(url, names) ترجع str
85    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
86    """Remove stale form values from a copied, already-submitted URL.
87 
88    A common way to configure a portal is to copy a URL that already contains
89    `?username=OLD&password=`.  Appending the next card gives two usernames;
90    many routers take the first one, so every attempt silently tests OLD.
91    Preserve routing/token parameters, but replace fields we submit ourselves.
92    """  # نهاية النص متعدد الأسطر
93    try:  # بدايةtry محمية (يليها except/finally)
94        url = portals.sanitize_login_url(url)  # إسناد نتيجة استدعاء portals.sanitize_login_url (معامل واحد) إلى url
95        parts = urlsplit(url)  # إسناد نتيجة استدعاء urlsplit (معامل واحد) إلى parts
96        remove = {str(n) for n in names if n}  # بناء اشتقاق مجموعة وإسناده إلى remove
97        query = [(k, v) for k, v in parse_qsl(parts.query,  # بناء اشتقاق قائمة وإسناده إلى query
98                                               keep_blank_values=True)  # المعامل المسمّى keep_blank_values
99                 if k not in remove]  # تكملة السطر السابق داخل القوس
100        return urlunsplit((parts.scheme, parts.netloc, parts.path,  # إرجاع urlunsplit(مجموعة)
101                           urlencode(query, doseq=True), parts.fragment))  # تكملة السطر السابق داخل القوس
102    # تكملة السطر السابق داخل القوس
103    except Exception:                                   # noqa: BLE001
104        return url  # إرجاع url

107def _get_request_url(p: dict, fields: dict) -> str:  # تعريف الدالة _get_request_url(p, fields) ترجع str
108    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
109    """Build GET query pairs in the order the copied/form URL established.
110 
111    Requests' `params=` appends fields in dictionary order (which used to put
112    password before username and added defaults before respecting the browser's
113    URL shape). Merge values into existing query slots, then append only truly
114    new fields.
115    """  # نهاية النص متعدد الأسطر
116    url = portals.sanitize_login_url(  # إسناد نتيجة استدعاء portals.sanitize_login_url (3 معاملات) إلى url
117        p.get("login_url", ""), p.get("user_field", "username"),  # عنصر في القائمة/المعاملات (يتبعه المزيد)
118        p.get("pass_field", "password"))  # تكملة السطر السابق داخل القوس
119    parts = urlsplit(url)  # إسناد نتيجة استدعاء urlsplit (معامل واحد) إلى parts
120    source = parse_qsl(parts.query, keep_blank_values=True)  # إسناد نتيجة استدعاء parse_qsl (معامل واحد، keep_blank_values=…) إلى source
121    submitted = {str(k): str(v) for k, v in fields.items()}  # بناء قاموس بالاشتقاق وإسناده إلى submitted
122    pairs, placed = [], set()  # إسناد مجموعة إلى مجموعة
123    for key, value in source:  # دورة على source باسم مجموعة
124        if key in submitted:  # شرط: key ضمن submitted
125            pairs.append((key, submitted[key]))  # استدعاء pairs.append (معامل واحد)
126            placed.add(key)  # استدعاء placed.add (معامل واحد)
127        else:  # مفتاح else في القاموس
128            pairs.append((key, value))  # استدعاء pairs.append (معامل واحد)
129    for key, value in fields.items():  # دورة على fields.items() باسم مجموعة
130        if str(key) not in placed:  # شرط: str(key) ليس ضمن placed
131            pairs.append((str(key), str(value)))  # استدعاء pairs.append (معامل واحد)
132    return urlunsplit((parts.scheme, parts.netloc, parts.path,  # إرجاع urlunsplit(مجموعة)
133                       urlencode(pairs, doseq=True), parts.fragment))  # تكملة السطر السابق داخل القوس

136def _request_components(p: dict, card: str):  # تعريف الدالة _request_components(p, card)
137    fields = build_fields(p, card)  # إسناد نتيجة استدعاء build_fields (2 معاملات) إلى fields
138    method = (p.get("method") or "post").lower()  # إسناد نتيجة استدعاء p.get('method') أو 'post'.lower إلى method
139    if method == "post":  # شرط: method يساوي 'post'
140        url = _without_query_fields(  # إسناد نتيجة استدعاء _without_query_fields (2 معاملات) إلى url

145    return method, url, fields  # إرجاع مجموعة

148def _safe_url_shape(url: str, p: dict) -> str:  # تعريف الدالة _safe_url_shape(url, p) ترجع str
149    parts = urlsplit(url or "")  # إسناد نتيجة استدعاء urlsplit (معامل واحد) إلى parts
150    safe_pairs = []  # إسناد قائمة إلى safe_pairs
151    user_name = (p.get("user_field") or "username").lower()  # إسناد نتيجة استدعاء p.get('user_field') أو 'username'.lower إلى user_name
152    pass_name = (p.get("pass_field") or "password").lower()  # إسناد نتيجة استدعاء p.get('pass_field') أو 'password'.lower إلى pass_name
153    for key, value in parse_qsl(parts.query, keep_blank_values=True):  # دورة على parse_qsl(parts.query, keep_blank_values=True) باسم مجموعة
154        low = key.lower()  # إسناد نتيجة استدعاء key.lower إلى low
155        if low == user_name:  # شرط: low يساوي user_name
156            shown = "[CARD]"  # إسناد القيمة الثابتة shown
157        elif low == pass_name:  # شرط: low يساوي pass_name
158            shown = "[EMPTY]" if not value else "[REDACTED]"  # إسناد '[EMPTY]' إن نفي/سالب وإلا '[REDACTED]' إلى shown
159        else:  # مفتاح else في القاموس
160            shown = "[REDACTED]" if value else "[EMPTY]"  # إسناد '[REDACTED]' إن value وإلا '[EMPTY]' إلى shown
161        safe_pairs.append((key, shown))  # استدعاء safe_pairs.append (معامل واحد)
162    # احذف userinfo من العنوان إضافة إلى قيم الاستعلام؛ فهو غير مطلوب
163    # لشرح شكل الطلب، وقد يحتوي بيانات دخول.
164    netloc = parts.netloc.rsplit("@", 1)[-1]  # إسناد parts.netloc.rsplit('@', 1)[نفي/سالب] إلى netloc
165    return urlunsplit((parts.scheme, netloc, parts.path,  # إرجاع urlunsplit(مجموعة)
166                       urlencode(safe_pairs, doseq=True), ""))  # تكملة السطر السابق داخل القوس

197def browser_headers(p: dict, url: str = "") -> dict:  # تعريف الدالة browser_headers(p, url) ترجع dict
198    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
199    """The headers a browser would send with this request.
200 
201    Some portals refuse anything that does not look like a browser: no
202    Referer, no Origin, no form content type, or a User-Agent they do not
203    like - and the operator then only ever sees "bad request".
204    """  # نهاية النص متعدد الأسطر
205    h = dict(config.BASE_HEADERS)  # إسناد نتيجة استدعاء dict (معامل واحد) إلى h
206    ua = (p.get("user_agent") or "").strip()  # إسناد نتيجة استدعاء p.get('user_agent') أو ''.strip إلى ua
207    if ua:  # شرط: ua
208        h["User-Agent"] = ua  # إسناد ua إلى h['User-Agent']
209    method = (p.get("method") or "post").lower()  # إسناد نتيجة استدعاء p.get('method') أو 'post'.lower إلى method
210    if url and p.get("send_referer", True):  # شرط مركّب (و)
211        # لا تصدّي بطاقة حقيقية من رابط منسوخ مُرسَل سابقاً إلى
212        # Referer. الطلب نفسه يستبدل تلك القيم أدناه أيضاً.
213        ref_url = _without_query_fields(  # إسناد نتيجة استدعاء _without_query_fields (2 معاملات) إلى ref_url
214            url, (p.get("user_field") or "username",  # عنصر في القائمة/المعاملات (يتبعه المزيد)
215                  p.get("pass_field") or "password"))  # تكملة السطر السابق داخل القوس
216        h["Referer"] = ref_url  # إسناد ref_url إلى h['Referer']
217        # المتصفح يرسل Origin مع POST لنموذج، لا مع تنقّل GET
218        # عادي. إرساله مع GET جعل رابط المتصفح المنسوخ يبدو أقل
219        # شبهاً بالمتصفح الذي نجح لتوّه مع المستخدم.
220        if method == "post":  # شرط: method يساوي 'post'
221            try:  # بدايةtry محمية (يليها except/finally)
222                parts = urlsplit(url)  # إسناد نتيجة استدعاء urlsplit (معامل واحد) إلى parts
223            # تكملة السطر السابق داخل القوس
224            except Exception:                           # noqa: BLE001
225                parts = None  # إسناد القيمة الثابتة parts
226            if parts and parts.scheme and parts.netloc:  # شرط مركّب (و)
227                h["Origin"] = f"{parts.scheme}://{parts.netloc}"  # بناء نص منسّق وإسناده إلى h['Origin']
228    if method == "post":  # شرط: method يساوي 'post'
229        h["Content-Type"] = "application/x-www-form-urlencoded"  # إسناد القيمة الثابتة h['Content-Type']
230    return h  # إرجاع h

242# يأخذ أسماء الحقول والرابط والقيم المخفية من صفحة طازجة.

244def absorb_form(p: dict, html: str, url: str) -> dict:  # تعريف الدالة absorb_form(p, html, url) ترجع dict
245    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
246    """Take what a freshly fetched login page is offering: field names, the
247    form action, and the hidden values (session token, dst, popup ...).
248 
249    Without this the tool keeps posting the values captured during the scan -
250    by another session, minutes ago - and portals that hand out a per-session
251    token answer "bad request" for every single card.
252    """  # نهاية النص متعدد الأسطر
253    if not (html or "").strip():  # شرط معكوس: ليس html أو ''.strip()
254        return p  # إرجاع p
255    form = portals.parse_form(html, url)  # إسناد نتيجة استدعاء portals.parse_form (2 معاملات) إلى form
256    if not form.inputs:  # شرط معكوس: ليس form.inputs
257        # صفحة بلا نموذج — صفحة حجب أو خطأ أو بوابة أدخلتنا
258        # فعلاً — يجب ألا تستبدل أبداً أسماء حقول نعرفها:
259        # parse_form() يسقط على username/password عندما لا يجد

300    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
301    """Ask for the login page first, the way a browser does.
302

304    per-session token, sometimes a chap challenge) *on the page*.  Posting
305    without them gets "bad request" instead of a judgement - which is why a
306    card that works in a browser never works from a script.  Returns the
307    profile refreshed with what the page just gave us, or **None** when the
308    portal did not give us a page at all: a post then would only be refused,
309    and the card must stay untested instead of paying for a wish.
310    """  # نهاية النص متعدد الأسطر
311    for attempt in range(max(1, int(tries))):  # دورة على range(max(1, int(tries))) باسم attempt
312        try:  # بدايةtry محمية (يليها except/finally)
313            resp = session.get(p["login_url"], allow_redirects=True)  # إسناد نتيجة استدعاء session.get (معامل واحد، allow_redirects=…) إلى resp

319        if attempt + 1 < max(1, int(tries)):  # شرط: جمع أصغر من max(1, int(tries))
320            time.sleep(0.15)  # استدعاء time.sleep (معامل واحد)
321    return None  # إرجاع None

425def bench_cards(p: dict, count: int = 3, seed: int = 0, exclude=()) -> list:  # تعريف الدالة bench_cards(p, count, seed, exclude) ترجع list
426    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
427    """Format-valid cards used to learn the rejection page.
428

430    true "wrong card" page - not with a "bad format" page, which was one of
431    the reasons the old baseline was wrong.
432    """  # نهاية النص متعدد الأسطر
433    space = store.space_size(p)  # إسناد نتيجة استدعاء store.space_size (معامل واحد) إلى space

435        return []  # إرجاع قائمة

437    wanted = min(max(0, int(count)), max(0, space - len(excluded)))  # إسناد نتيجة استدعاء min (2 معاملات) إلى wanted
438    if not wanted:  # شرط معكوس: ليس wanted
439        return []  # إرجاع قائمة

442    attempts = 0  # إسناد القيمة الثابتة attempts
443    while len(picks) < wanted and attempts < max(space * 2, wanted * 20):  # حلقة ما دام مقارنة و مقارنة
444        idx = rnd.randrange(space)  # إسناد نتيجة استدعاء rnd.randrange (معامل واحد) إلى idx

449    if len(picks) < wanted and space <= 100000:  # شرط مركّب (و)

452                picks.add(idx)  # استدعاء picks.add (معامل واحد)
453                if len(picks) >= wanted:  # شرط: len(picks) أكبر أو يساوي wanted
