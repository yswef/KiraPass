1# بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
2"""Read a captive-portal login page and copy exactly what the browser sends.
3 
4No questions are asked here: the page itself says which URL the form posts to,
5which field names are used, which hidden values (dst / popup / chap-id) must
6travel with the request.  MikroTik's `md5.js` trick is detected too, because
7most custom card templates use it:
8 
9    document.login.password.value = hexMD5('$(chap-id)' + password +
10                                          '$(chap-challenge)')
11 
12Sending the raw card instead of that hash is a silent killer: the portal
13rejects everything, and the tool looks broken.
14"""  # نهاية النص متعدد الأسطر
15 
16from __future__ import annotations  # استيراد annotations من الوحدة __future__
17 
18import html as htmlmod  # استيراد الوحدة html من المكتبة
19import re  # استيراد الوحدة re من المكتبة
20from urllib.parse import parse_qs, parse_qsl, urlencode, urljoin, urlsplit, urlunsplit  # استيراد parse_qs, parse_qsl, urlencode, urljoin, urlsplit, urlunsplit من الوحدة urllib.parse
21 
22FORM_RE = re.compile(r"<form\b[^>]*>", re.I)  # إسناد نتيجة استدعاء re.compile (2 معاملات) إلى FORM_RE
23INPUT_RE = re.compile(r"<input\b[^>]*>", re.I)  # إسناد نتيجة استدعاء re.compile (2 معاملات) إلى INPUT_RE
24ATTR_RE = re.compile(r"([\w\-:]+)\s*=\s*(\"[^\"]*\"|'[^']*'|[^\s>]+)")  # إسناد نتيجة استدعاء re.compile (معامل واحد) إلى ATTR_RE
25CHAP_CALL_RE = re.compile(  # إسناد نتيجة استدعاء re.compile (2 معاملات) إلى CHAP_CALL_RE
26    r"hexMD5\s*\(\s*'([^']*)'\s*\+\s*([^+)]+?)\s*\+\s*'([^']*)'\s*\)", re.I)  # تكملة السطر السابق داخل القوس
27CHAP_SIMPLE_RE = re.compile(r"hexMD5\s*\(\s*([^)]+)\)", re.I)  # إسناد نتيجة استدعاء re.compile (2 معاملات) إلى CHAP_SIMPLE_RE
28TITLE_RE = re.compile(r"<title[^>]*>(.*?)</title>", re.I | re.S)  # إسناد نتيجة استدعاء re.compile (2 معاملات) إلى TITLE_RE
29 
30# حقول يجب ألا ننسخها أبداً كـ«إضافات ثابتة» (هي تحمل قيمنا نحن).
31CORE_FIELDS = ("username", "user", "password", "passwd", "pass", "card",  # إسناد مجموعة إلى CORE_FIELDS
32               "code", "voucher", "dst", "popup")  # تكملة السطر السابق داخل القوس
33

35def _attrs(tag: str) -> dict:  # تعريف الدالة _attrs(tag) ترجع dict
36    out = {}  # إسناد قاموس إلى out
37    for name, value in ATTR_RE.findall(tag):  # دورة على ATTR_RE.findall(tag) باسم مجموعة
38        value = value.strip()  # إسناد نتيجة استدعاء value.strip إلى value
39        if len(value) >= 2 and value[0] in "\"'" and value[-1] == value[0]:  # شرط مركّب (و)
40            value = value[1:-1]  # إسناد value[] إلى value
41        # HTML يرمّز علامة & كـ&amp; داخل الخصائص. فكّك الترميز قبل
42        # ضم/تحليل الرابط، وإلا تغيّر شكل رابط المتصفح.
43        out[name.lower()] = htmlmod.unescape(value)  # إسناد نتيجة استدعاء htmlmod.unescape (معامل واحد) إلى out[name.lower()]
44    return out  # إرجاع out

47class FormInfo:  # تعريف الصنف FormInfo
48    def __init__(self, action, method, fields, inputs):  # تعريف الدالة __init__(self, action, method, fields, inputs)
49        self.action = action  # إسناد action إلى self.action
50        self.method = method  # إسناد method إلى self.method
51        self.fields = fields          # الاسم ← القيمة الافتراضية
52        self.inputs = inputs          # قائمة (الاسم، النوع)
53        self.user_field = ""  # إسناد القيمة الثابتة self.user_field
54        self.pass_field = ""  # إسناد القيمة الثابتة self.pass_field
55        self.dst_field = ""  # إسناد القيمة الثابتة self.dst_field
56        self.dst_value = ""  # إسناد القيمة الثابتة self.dst_value
57        self.popup_field = ""  # إسناد القيمة الثابتة self.popup_field
58        self.extra_fields = {}        # حقول مخفية تُرسل دائماً
59        self.chap = None              # {'id':..,'challenge':..,'field':'password'}
60 
61    @property  # مُزخرف (decorator) بـproperty
62    def is_post(self) -> bool:  # تعريف الدالة is_post(self) ترجع bool
63        return self.method.lower() == "post"  # إرجاع مقارنة
64 
65    def as_dict(self) -> dict:  # تعريف الدالة as_dict(self) ترجع dict
66        return {  # إرجاع قاموس
67            "action": self.action,  # مفتاح action في القاموس
68            "method": self.method,  # مفتاح method في القاموس
69            "user_field": self.user_field,  # مفتاح user_field في القاموس
70            "pass_field": self.pass_field,  # مفتاح pass_field في القاموس
71            "dst_field": self.dst_field,  # مفتاح dst_field في القاموس
72            "dst_value": self.dst_value,  # مفتاح dst_value في القاموس
73            "popup_field": self.popup_field,  # مفتاح popup_field في القاموس
74            "extra_fields": self.extra_fields,  # مفتاح extra_fields في القاموس
75            "all_fields": list(self.fields.keys()),  # مفتاح all_fields في القاموس
76            "chap": self.chap or None,  # مفتاح chap في القاموس
77        }  # إغلاق القوس المفتوح في السطر السابق

80def _form_segments(html: str) -> list:  # تعريف الدالة _form_segments(html) ترجع list
81    """-> [(form tag, html inside that form)] for every <form> on the page."""  # نص توثيقي (docstring) يشرح ما يليه
82    out = []  # إسناد قائمة إلى out
83    marks = list(FORM_RE.finditer(html))  # إسناد نتيجة استدعاء list (معامل واحد) إلى marks
84    for i, m in enumerate(marks):  # دورة على enumerate(marks) باسم مجموعة
85        start = m.end()  # إسناد نتيجة استدعاء m.end إلى start
86        close = html.lower().find("</form", start)  # إسناد نتيجة استدعاء html.lower().find (2 معاملات) إلى close
87        end = marks[i + 1].start() if i + 1 < len(marks) else len(html)  # إسناد marks[جمع].start() إن مقارنة وإلا len(html) إلى end
88        if close != -1:  # شرط: close لا يساوي نفي/سالب
89            end = min(end, close)  # إسناد نتيجة استدعاء min (2 معاملات) إلى end
90        out.append((m.group(0), html[start:max(end, start)]))  # استدعاء out.append (معامل واحد)
91    return out  # إرجاع out

94def _score_form(attrs: dict, segment: str) -> int:  # تعريف الدالة _score_form(attrs, segment) ترجع int
95    """How much does this form look like the login form? (higher = better)."""  # نص توثيقي (docstring) يشرح ما يليه
96    score = 0  # إسناد القيمة الثابتة score
97    tags = [_attrs(t) for t in INPUT_RE.findall(segment)]  # بناء اشتقاق قائمة وإسناده إلى tags
98    types = [(a.get("type") or "text").lower() for a in tags]  # بناء اشتقاق قائمة وإسناده إلى types
99    names = [(a.get("name") or "").lower() for a in tags]  # بناء اشتقاق قائمة وإسناده إلى names
100    if "password" in types:  # شرط: 'password' ضمن types
101        score += 5  # تحديث score بعملية جمع
102    if any(t in ("text", "tel", "number", "email") for t in types):  # شرط: نتيجة any(مولّد)
103        score += 2  # تحديث score بعملية جمع
104    if any(w in n for n in names for w in ("user", "pass", "card", "voucher",  # شرط: نتيجة any(مولّد)
105                                           "code", "pin")):  # تكملة تعريف متعدد الأسطر
106        score += 2  # تحديث score بعملية جمع
107    blob = " ".join((attrs.get("name") or "", attrs.get("id") or "",  # إسناد نتيجة استدعاء ' '.join(مجموعة).lower إلى blob
108                     attrs.get("action") or "")).lower()  # تكملة السطر السابق داخل القوس
109    if any(w in blob for w in ("login", "logon", "auth", "hotspot", "portal",  # شرط: نتيجة any(مولّد)
110                               "signin", "sign-in", "connect")):  # تكملة تعريف متعدد الأسطر
111        score += 3  # تحديث score بعملية جمع
112    if (attrs.get("method") or "").lower() == "post":  # شرط: attrs.get('method') أو ''.lower() يساوي 'post'
113        score += 1  # تحديث score بعملية جمع
114    return score  # إرجاع score

117# يختار نموذج الدخول من بين نماذج الصفحة بأعلى نتيجة، لا أول نموذج.
118# يسقط على username/password إن لم يجد أسماء أوضح.
119def parse_form(html: str, base_url: str) -> FormInfo:  # تعريف الدالة parse_form(html, base_url) ترجع FormInfo
120    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
121    """Find the login form and classify every field in it.
122 
123    A captive-portal page often carries more than one form (a language picker,
124    a search box, a "buy a card" form).  Taking the first <form> blindly used
125    to fill the request with the wrong field names, so every form is scored
126    and the most login-looking one wins; fields are then read from inside it.
127    """  # نهاية النص متعدد الأسطر
128    html = html or ""  # دمج منطقي (أو) وإسناده إلى html
129    base_url = htmlmod.unescape(str(base_url or ""))  # إسناد نتيجة استدعاء htmlmod.unescape (معامل واحد) إلى base_url
130    action, method = base_url, "post"  # إسناد مجموعة إلى مجموعة
131    body = html  # إسناد html إلى body
132 
133    segments = _form_segments(html)  # إسناد نتيجة استدعاء _form_segments (معامل واحد) إلى segments
134    # رابط متصفح ناجح منسوخ غالباً *هو* إرسال النموذج:
135    # /login?username=293...&password=
136    # إن لم يكن في الصفحة نموذج HTML، فذلك الاستعلام دليل مباشر على GET —
137    # لا تسقط على POST ولا تضف username ثانياً لاحقاً.
138    query_pairs = parse_qsl(urlsplit(base_url).query, keep_blank_values=True)  # إسناد نتيجة استدعاء parse_qsl (معامل واحد، keep_blank_values=…) إلى query_pairs
139    query_names = {name.lower() for name, _ in query_pairs}  # بناء اشتقاق مجموعة وإسناده إلى query_names
140    query_login = (not segments and  # دمج منطقي (و) وإسناده إلى query_login
141                   any(any(w in name for w in  # تكملة السطر السابق داخل القوس
142                           ("user", "login", "card", "voucher", "account"))  # تكملة السطر السابق داخل القوس
143                       for name in query_names) and  # تكملة السطر السابق داخل القوس
144                   any(any(w in name for w in ("pass", "pwd", "pin"))  # تكملة السطر السابق داخل القوس
145                       for name in query_names))  # تكملة السطر السابق داخل القوس
146    if query_login:  # شرط: query_login
147        method = "get"  # إسناد القيمة الثابتة method
148    if segments:  # شرط: segments
149        best = max(segments, key=lambda pair: _score_form(_attrs(pair[0]),  # إسناد نتيجة استدعاء max (معامل واحد، key=…) إلى best
150                                                          pair[1]))  # تكملة السطر السابق داخل القوس
151        a = _attrs(best[0])  # إسناد نتيجة استدعاء _attrs (معامل واحد) إلى a
152        if a.get("action"):  # شرط: نتيجة a.get('action')
153            action = urljoin(base_url, a["action"])  # إسناد نتيجة استدعاء urljoin (2 معاملات) إلى action
154        if a.get("method"):  # شرط: نتيجة a.get('method')
155            method = a["method"].lower()  # إسناد نتيجة استدعاء a['method'].lower إلى method
156        if INPUT_RE.search(best[1]):  # شرط: نتيجة INPUT_RE.search(best[1])
157            body = best[1]  # إسناد best[1] إلى body
158 
159    fields, inputs = {}, []  # إسناد مجموعة إلى مجموعة
160    for tag in INPUT_RE.findall(body):  # دورة على INPUT_RE.findall(body) باسم tag
161        a = _attrs(tag)  # إسناد نتيجة استدعاء _attrs (معامل واحد) إلى a
162        name = a.get("name")  # إسناد نتيجة استدعاء a.get (معامل واحد) إلى name
163        if not name:  # شرط معكوس: ليس name
164            continue  # الانتقال إلى الدورة التالية
165        fields[name] = a.get("value", "")  # إسناد نتيجة استدعاء a.get (2 معاملات) إلى fields[name]
166        inputs.append((name, a.get("type", "text").lower()))  # استدعاء inputs.append (معامل واحد)
167 
168    if not inputs and body is not html:  # شرط مركّب (و)
169        # صفحة مشوّهة: وُجد وسم النموذج لكن الحقول خارجه
170        for tag in INPUT_RE.findall(html):  # دورة على INPUT_RE.findall(html) باسم tag
171            a = _attrs(tag)  # إسناد نتيجة استدعاء _attrs (معامل واحد) إلى a
172            name = a.get("name")  # إسناد نتيجة استدعاء a.get (معامل واحد) إلى name
173            if not name:  # شرط معكوس: ليس name
174                continue  # الانتقال إلى الدورة التالية
175            fields[name] = a.get("value", "")  # إسناد نتيجة استدعاء a.get (2 معاملات) إلى fields[name]
176            inputs.append((name, a.get("type", "text").lower()))  # استدعاء inputs.append (معامل واحد)
177 
178    if query_login:  # شرط: query_login
179        existing = {name for name, _type in inputs}  # بناء اشتقاق مجموعة وإسناده إلى existing
180        for name, value in query_pairs:  # دورة على query_pairs باسم مجموعة
181            fields.setdefault(name, value)  # استدعاء fields.setdefault (2 معاملات)
182            if name in existing:  # شرط: name ضمن existing
183                continue  # الانتقال إلى الدورة التالية
184            low = name.lower()  # إسناد نتيجة استدعاء name.lower إلى low
185            typ = ("password" if any(w in low for w in ("pass", "pwd", "pin"))  # إسناد 'password' إن any(مولّد) وإلا 'text' إن any(مولّد) وإلا 'hidden' إلى typ
186                   else "text" if any(w in low for w in  # تكملة السطر السابق داخل القوس
187                                      ("user", "login", "card", "voucher",  # عنصر في القائمة/المعاملات (يتبعه المزيد)
188                                       "account"))  # تكملة السطر السابق داخل القوس
189                   else "hidden")  # تكملة السطر السابق داخل القوس
190            inputs.append((name, typ))  # استدعاء inputs.append (معامل واحد)
191 
192    info = FormInfo(action, method, fields, inputs)  # إسناد نتيجة استدعاء FormInfo (4 معاملات) إلى info
193 
194    # حقل كلمة المرور أولاً (type=password يفوز، ثم الاسم)
195    # دمج منطقي (أو) وإسناده إلى info.pass_field
196    info.pass_field = _pick(inputs, lambda n, t: t == "password") or \
197        _pick(inputs, lambda n, t: any(w in n.lower() for w in  # تكملة السطر السابق داخل القوس
198                                       ("pass", "pwd", "pin", "voucher", "code"))  # تكملة السطر السابق داخل القوس
199              and t not in ("hidden", "submit")) or "password"  # تكملة السطر السابق داخل القوس
200    info.user_field = _pick(inputs, lambda n, t: n != info.pass_field  # دمج منطقي (أو) وإسناده إلى info.user_field
201                            and any(w in n.lower() for w in  # تكملة السطر السابق داخل القوس
202                                    ("user", "card", "code", "voucher", "login",  # عنصر في القائمة/المعاملات (يتبعه المزيد)
203                                     "account", "name"))  # تكملة السطر السابق داخل القوس
204                            and t not in ("hidden", "submit", "button", "reset",  # عنصر في القائمة/المعاملات (يتبعه المزيد)
205                                          "checkbox", "radio", "image",  # عنصر في القائمة/المعاملات (يتبعه المزيد)
206                                          # تكملة السطر السابق داخل القوس
207                                          "file")) or \
208        _pick(inputs, lambda n, t: n != info.pass_field  # تكملة السطر السابق داخل القوس
209              and t in ("text", "tel", "number", "email")) or "username"  # تكملة السطر السابق داخل القوس
210 
211    # الأزرار ليست بيانات: المتصفح يرسل فقط الزر المضغوط، و
212    # إرسال كل زر مسمّى في الصفحة يربك بعض البوابات
213    NON_DATA_TYPES = ("submit", "button", "reset", "image", "file")  # إسناد مجموعة إلى NON_DATA_TYPES
214    for name, _type in inputs:  # دورة على inputs باسم مجموعة
215        low = name.lower()  # إسناد نتيجة استدعاء name.lower إلى low
216        if low in ("dst", "link-orig"):  # شرط: low ضمن مجموعة
217            info.dst_field, info.dst_value = name, fields.get(name, "")  # إسناد مجموعة إلى مجموعة
218            continue  # الانتقال إلى الدورة التالية
219        if _type in NON_DATA_TYPES:  # شرط: _type ضمن NON_DATA_TYPES
220            continue  # الانتقال إلى الدورة التالية
221        if low == "popup":  # شرط: low يساوي 'popup'
222            info.popup_field = name  # إسناد name إلى info.popup_field
223        elif low not in (info.user_field.lower(), info.pass_field.lower(),  # شرط: low ليس ضمن مجموعة
224                         info.dst_field.lower()):  # تكملة تعريف متعدد الأسطر
225            info.extra_fields[name] = fields.get(name, "")  # إسناد نتيجة استدعاء fields.get (2 معاملات) إلى info.extra_fields[name]
226    if not info.dst_field:  # شرط معكوس: ليس info.dst_field
227        info.dst_field = "dst"  # إسناد القيمة الثابتة info.dst_field
228    if not info.popup_field:  # شرط معكوس: ليس info.popup_field
229        info.popup_field = "popup"  # إسناد القيمة الثابتة info.popup_field
230 
231    # احفظ رابطاً بنفس مفاتيح/ترتيب الاستعلام لكن بقيم فارغة للمستخدم/كلمة المرور.
232    # كود إرسال GET يمدّ القيم الطازجة لكل محاولة.
233    info.action = sanitize_login_url(info.action, info.user_field, info.pass_field)  # إسناد نتيجة استدعاء sanitize_login_url (3 معاملات) إلى info.action
234 
235    # chap في ميكروتيك: hexMD5('id' + كلمة المرور + 'challenge')
236    cm = CHAP_CALL_RE.search(html)  # إسناد نتيجة استدعاء CHAP_CALL_RE.search (معامل واحد) إلى cm
237    if cm:  # شرط: cm
238        info.chap = {"id": cm.group(1), "challenge": cm.group(3),  # إسناد قاموس إلى info.chap
239                     "field": info.pass_field}  # مفتاح field في القاموس
240    elif "hexMD5" in html:  # شرط: 'hexMD5' ضمن html
241        info.chap = {"id": "", "challenge": "", "field": info.pass_field}  # إسناد قاموس إلى info.chap
242    return info  # إرجاع info

245def _pick(inputs, predicate) -> str:  # تعريف الدالة _pick(inputs, predicate) ترجع str
246    for name, typ in inputs:  # دورة على inputs باسم مجموعة
247        try:  # بدايةtry محمية (يليها except/finally)
248            if predicate(name, typ):  # شرط: نتيجة predicate(name, typ)
249                return name  # إرجاع name

282# يحذف قيم username/password من رابط منسوخ قبل حفظه أو عرضه.
283def sanitize_login_url(url: str, user_field: str = "username",  # تعريف الدالة sanitize_login_url(url, user_field, pass_field) ترجع str
284                       pass_field: str = "password") -> str:  # مفتاح pass_field في القاموس
285    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
286    """Canonicalize a login URL and remove credential values from its query.
287 
288    Keep the query keys and their order (some portals depend on the URL shape),
289    but never persist a card or password/hash copied from a submitted GET link.
290    """  # نهاية النص متعدد الأسطر
291    value = htmlmod.unescape(str(url or "")).strip()  # إسناد نتيجة استدعاء htmlmod.unescape(str(url أو '')).strip إلى value
292    try:  # بدايةtry محمية (يليها except/finally)
293        parts = urlsplit(value)  # إسناد نتيجة استدعاء urlsplit (معامل واحد) إلى parts
294        secret_keys = {str(user_field or "username").lower(),  # إسناد مجموعة فريدة إلى secret_keys
295                       str(pass_field or "password").lower()}  # تكملة السطر السابق داخل القوس
296        query = [(key, "" if key.lower() in secret_keys else item)  # بناء اشتقاق قائمة وإسناده إلى query
297                 for key, item in parse_qsl(parts.query, keep_blank_values=True)]  # تكملة السطر السابق داخل القوس
298        return urlunsplit((parts.scheme, parts.netloc, parts.path,  # إرجاع urlunsplit(مجموعة)
299                           urlencode(query, doseq=True), parts.fragment))  # تكملة السطر السابق داخل القوس
300    except Exception:  # تكملة السطر السابق داخل القوس
301        return value  # إرجاع value

304def _copied_login_query(url: str) -> tuple:  # تعريف الدالة _copied_login_query(url) ترجع tuple
305    """(looks like a submitted GET login, credential field names)."""  # نص توثيقي (docstring) يشرح ما يليه
306    url = htmlmod.unescape(str(url or ""))  # إسناد نتيجة استدعاء htmlmod.unescape (معامل واحد) إلى url
307    pairs = parse_qsl(urlsplit(url).query, keep_blank_values=True)  # إسناد نتيجة استدعاء parse_qsl (معامل واحد، keep_blank_values=…) إلى pairs
308    names = {name.lower(): name for name, _ in pairs}  # بناء قاموس بالاشتقاق وإسناده إلى names
309    users = {original for low, original in names.items()  # بناء اشتقاق مجموعة وإسناده إلى users
310             if any(w in low for w in  # تكملة السطر السابق داخل القوس
311                    ("user", "login", "card", "voucher", "account"))}  # تكملة السطر السابق داخل القوس
312    passwords = {original for low, original in names.items()  # بناء اشتقاق مجموعة وإسناده إلى passwords
313                 if any(w in low for w in ("pass", "pwd", "pin"))}  # تكملة السطر السابق داخل القوس
314    return bool(users and passwords), users | passwords  # إرجاع مجموعة

317def _safe_page_url(url: str) -> str:  # تعريف الدالة _safe_page_url(url) ترجع str
318    """Open the login *page*, never spend the card embedded in a copied URL."""  # نص توثيقي (docstring) يشرح ما يليه
319    url = htmlmod.unescape(str(url or "")).strip()  # إسناد نتيجة استدعاء htmlmod.unescape(str(url أو '')).strip إلى url
320    copied, credential_names = _copied_login_query(url)  # إسناد نتيجة استدعاء _copied_login_query (معامل واحد) إلى مجموعة
321    if not copied:  # شرط معكوس: ليس copied
322        return url  # إرجاع url
323    parts = urlsplit(url)  # إسناد نتيجة استدعاء urlsplit (معامل واحد) إلى parts
324    pairs = [(k, v) for k, v in parse_qsl(parts.query, keep_blank_values=True)  # بناء اشتقاق قائمة وإسناده إلى pairs
325             if k not in credential_names]  # تكملة السطر السابق داخل القوس
326    return urlunsplit((parts.scheme, parts.netloc, parts.path,  # إرجاع urlunsplit(مجموعة)
327                       urlencode(pairs, doseq=True), parts.fragment))  # تكملة السطر السابق داخل القوس

330def discover(session, url: str, timeout=None) -> Portal:  # تعريف الدالة discover(session, url, timeout) ترجع Portal
331    # بداية نص متعدد الأسطر — الأسطر التالية نصّ حرفي لا كود، فلا يقبل تعليقاً
332    """Open the login page and report everything the tool learned from it.
333 
334    If the operator pasted an already-submitted GET URL, remove its username
335    and password before opening it: scanning must never consume a real card.
336    The original query still teaches us that this is a GET portal.
337    """  # نهاية النص متعدد الأسطر
338    url = htmlmod.unescape(str(url or "")).strip()  # إسناد نتيجة استدعاء htmlmod.unescape(str(url أو '')).strip إلى url
339    page_url = _safe_page_url(url)  # إسناد نتيجة استدعاء _safe_page_url (معامل واحد) إلى page_url
340    resp = session.get(page_url, allow_redirects=True, timeout=timeout)  # إسناد نتيجة استدعاء session.get (معامل واحد، allow_redirects=…، timeout=…) إلى resp
341    base = resp.url or page_url  # دمج منطقي (أو) وإسناده إلى base
342    form = parse_form(resp.text, base)  # إسناد نتيجة استدعاء parse_form (2 معاملات) إلى form
343    copied_get, _credential_names = _copied_login_query(url)  # إسناد نتيجة استدعاء _copied_login_query (معامل واحد) إلى مجموعة
344    if copied_get and not _form_segments(resp.text or ""):  # شرط مركّب (و)
345        form = parse_form(resp.text, url)  # إسناد نتيجة استدعاء parse_form (2 معاملات) إلى form
346        form.action = base  # إسناد base إلى form.action
347 
348    dsts, seen = [], set()  # إسناد مجموعة إلى مجموعة
349    for src in (parse_qs(urlsplit(base).query).get("dst", [None])[0],  # دورة على مجموعة باسم src
350                form.dst_value if form else None,  # عنصر في القائمة/المعاملات (يتبعه المزيد)
351                _js_var(resp.text, "link-orig"),  # عنصر في القائمة/المعاملات (يتبعه المزيد)
352                ""):  # تكملة تعريف متعدد الأسطر
353        if src is not None and src not in seen:  # شرط مركّب (و)
354            seen.add(src)  # استدعاء seen.add (معامل واحد)
355            dsts.append(src)  # استدعاء dsts.append (معامل واحد)
356 
357    notes = []  # إسناد قائمة إلى notes
358    if form and form.chap:  # شرط مركّب (و)
359        notes.append("chap_md5_detected")  # استدعاء notes.append (معامل واحد)
360    if form and form.is_post:  # شرط مركّب (و)
361        notes.append("post_form")  # استدعاء notes.append (معامل واحد)
362    else:  # مفتاح else في القاموس
363        notes.append("get_form")  # استدعاء notes.append (معامل واحد)
364    if not form or not form.fields:  # شرط مركّب (أو)
365        notes.append("no_form_found")  # استدعاء notes.append (معامل واحد)
366    if resp.history:  # شرط: resp.history
367        notes.append("page_redirected_here")  # استدعاء notes.append (معامل واحد)
368 
369    return Portal(base, resp, form, dsts, notes)  # إرجاع Portal(base, resp, form, …)

372def _js_var(html: str, name: str) -> str:  # تعريف الدالة _js_var(html, name) ترجع str
373    m = re.search(rf"{re.escape(name)}\s*[:=]\s*[\"']([^\"']*)[\"']", html or "", re.I)  # إسناد نتيجة استدعاء re.search (3 معاملات) إلى m
374    return m.group(1) if m else ""  # إرجاع m.group(1) إن m وإلا ''
